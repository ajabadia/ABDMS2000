19enum class DiagnosticTonePoint {
20 Disabled = 0,
21 PostMasterVolume = 1, // At the very end of processBlock (after master volume)
22 PreMasterVolume = 2, // Before master volume attenuation
23 PreEffects = 3, // After Voices/Vocoder, before ModFX / DelayFX / MasterEQ
24 PreFilter = 4, // Inside Voice: replaces mixedAudio before filter_
25 OscillatorDirect = 5 // Inside Voice: replaces OSC1 output before mixer & VCF
26};
