20/**
21 * @brief Programa de fábrica del Korg MS2000 / MS2000R / microKORG — 254 bytes reales.
22 *
23 * Este es el **formato del equipo físico**, no el del plugin. El plugin guarda sus
24 * presets propios en `MS2000ProgramData` (bloque nativo del motor); esto de
25 * aquí es lo que sale y entra por MIDI de un MS2000 de verdad.
26 *
27 * Mapa verificado contra un dump real (patch "INIT Program" del panel ReMS2000,
28 * `DOCS/ReMS2000-main/ReMS2000.panel` → `initPatchData`, contrastado con
29 * `applyProgramData` / `applyTimbreData` / `applyVocoderData` del mismo panel):
30 *
31 * 0x00..0x0B Nombre (12 B, ASCII) ← el nombre va AQUÍ, no en 0x1C
32 * 0x0C..0x0F Sin uso
33 * 0x10 Timbre Voice (bits 6,7) + Voice Mode (bits 4,5: 0 Single, 1 Split, 2 Layer, 3 Vocoder)
34 * 0x11 Scale Key (bits 4..7) + Scale Type (bits 0..3)
35 * 0x12 Split Point
36 * 0x13..0x16 Delay FX: flags (bit 7 tempo sync, bits 0..3 time base), time, feedback, type
37 * 0x17..0x19 Mod FX: speed, depth, type
38 * 0x1A..0x1D EQ: high freq, high gain (+64), low freq, low gain (+64)
39 * 0x1E..0x24 Arpegiador: tempo MSB/LSB, flags, type+range, gate, resolution, swing
40 * 0x25 Sin uso
41 * 0x26..0x91 TIMBRE 1 (108 B) → ver `ti`
42 * 0x92..0xFD TIMBRE 2 (108 B) → mismo layout que Timbre 1
43 *
44 * El bloque de un timbre (108 B) tiene un significado distinto según el Voice Mode:
45 * con `Vocoder` (3) sus bytes 15..29 y 46..77 son las bandas del vocoder
46 * (`knobSeq1Step`/`knobSeq2Step` en ReMS2000 = nivel y pan de banda), no los
47 * osciladores/filtro del timbre. Por eso hay dos tablas de mapeo al motor.
48 *
49 * La trama en el cable es `F0 42 3n 58 40 [254 B empaquetados 7→8] F7`; el
50 * empaquetado de Korg **no rellena** el último grupo parcial: 254 B → 291 B
51 * (`abd::hw::SysExCodec` ya lo hace así, a diferencia del codec de ABDBankManager).
52 *
53 * Qué hace con el motor (`applyToAPVTS`) y qué no: ver `unmodelled()`.
54 */

225 /** Desempaqueta 254 B de un payload 7→8 real (291 B; tolera el último grupo parcial). */
226 bool unpackFromSysexPayload(const uint8_t* payload, size_t payloadLen)
227 {
228 std::vector<uint8_t> unpacked;
229 if (!SysExCodec::unpack7to8(payload, payloadLen, unpacked) || unpacked.empty())
230 return false;
231 if (unpacked.size() < PROGRAM_SIZE) return false;
232 std::copy_n(unpacked.begin(), PROGRAM_SIZE, raw.begin());
233 return true;
234 }

242 /**
243 * Byte `3n` de la cabecera de Korg: `0x30 | (canal - 1)` (canal 1 -> `0x30`).
244 *
245 * Convención consolidada con el contrato del Bank Manager: canal 1 -> `0x30`
246 * y canal 16 -> `0x3F`. La fórmula está compartida por las tramas C++ y TS;
247 * cualquier validación frente a un MS2000 real queda documentada en ROADMAP §4.
248 */
249 static uint8_t korgChannelByte(int channel)
250 {
251 const int ch = std::max(1, std::min(16, channel));
252 return static_cast<uint8_t>(0x30 | (ch - 1));
253 }

255 /** Trama completa `F0 42 3n 58 40 [payload] F7`. */
256 std::vector<uint8_t> buildProgramDump(int channel = 1, uint8_t function = 0x40) const
257 {
258 std::vector<uint8_t> packed;
259 if (!packToSysexPayload(packed)) return {};
260
261 std::vector<uint8_t> sysex;
262 sysex.reserve(6 + packed.size());
263 sysex.push_back(0xF0);
264 sysex.push_back(0x42); // Korg
265 sysex.push_back(korgChannelByte(channel));
266 sysex.push_back(0x58); // MS2000 / MS2000R
267 sysex.push_back(function); // 0x40 = Program Data Dump
268 sysex.insert(sysex.end(), packed.begin(), packed.end());
269 sysex.push_back(0xF7);
270 return sysex;
271 }

276 // ─── All Data Dump (`0x4C`): la memoria completa del equipo ──────────────
277
278 /**
279 * Comando del volcado completo de la memoria (`MS2000_SysEx_Spec.md` §2).
280 * Se llama `CMD_` por el mismo motivo que los acuses: `0x4C` no es un offset.
281 */
282 static constexpr uint8_t CMD_ALL_DATA_DUMP = 0x4C;
283
284 /**
285 * Longitud del payload empaquetado de un All Data Dump de `count` programas.
286 *
287 * El empaquetado de Korg se aplica al **flujo completo** (los `count × 254` B
288 * concatenados), no a cada programa por separado. La diferencia importa: 254 no es
289 * múltiplo de 7, así que empaquetar cada programa y concatenar los trozos deja los
290 * bytes de MSBs en otro sitio y el receptor descodifica otra cosa. Con 128 programas
291 * el flujo son 32 512 B → 37 157 B de payload (y 37 163 B de trama), mientras que
292 * 128 trozos independientes darían 37 248 B.
293 *
294 * ⚠️ Hipótesis **pendiente de verificar contra hardware real**: el receptor de este
295 * repo (`SysExManager`, comando `0x4C`) deshace el volcado como flujo único, que es
296 * lo coherente con esto, pero nadie ha grabado todavía un All Data Dump del MS2000
297 * para confirmarlo (ver el roadmap).
298 */
299 static size_t allDataDumpPayloadSize(size_t count)
300 {
301 const size_t total = count * PROGRAM_SIZE;
302 const size_t remainder = total % 7;
303 return ((total / 7) * 8) + (remainder == 0 ? 0 : remainder + 1);
304 }

306 /**
307 * Trama de All Data Dump: `F0 42 3n 58 4C [count × 254 B empaquetados] F7`.
308 *
309 * Es la memoria completa del equipo (los 128 programas de la máquina) con la
310 * cabecera de Korg. Devuelve vacío si no hay programas: un volcado sin memoria no
311 * significa nada y no se manda.
312 */
313 static std::vector<uint8_t> buildAllDataDump(int channel,
314 const std::vector<MS2000HardwareProgram>& programs)
315 {
316 if (programs.empty()) return {};
317
318 std::vector<uint8_t> unpacked;
319 unpacked.reserve(programs.size() * PROGRAM_SIZE);
320 for (const auto& program : programs)
321 unpacked.insert(unpacked.end(), program.raw.begin(), program.raw.end());
322
323 std::vector<uint8_t> packed;
324 if (!SysExCodec::pack8to7(unpacked.data(), unpacked.size(), packed)) return {};
325
326 std::vector<uint8_t> sysex;
327 sysex.reserve(6 + packed.size());
328 sysex.push_back(0xF0);
329 sysex.push_back(0x42); // Korg
330 sysex.push_back(korgChannelByte(channel));
331 sysex.push_back(0x58); // MS2000 / MS2000R
332 sysex.push_back(CMD_ALL_DATA_DUMP);
333 sysex.insert(sysex.end(), packed.begin(), packed.end());
334 sysex.push_back(0xF7);
335 return sysex;
336 }

354 /**
355 * Cómo se convierte un byte/bit del dump al valor del parámetro del motor.
356 *
357 * El Timbre 2 usa **la misma tabla** que el Timbre 1: el programa real guarda dos
358 * bloques de 108 B idénticos en estructura (0x26 y 0x92), y el id del parámetro se
359 * resuelve cambiando el prefijo (`osc1Wave` → `t2Osc1Wave`).
360 */
361 enum class Xform : uint8_t
362 {
363 Raw, // 0..127 igual
364 Signed, // byte - 64
365 Bit, // bit suelto
366 EqFreq, // 0..29 real → 0..3 del motor (aproximación declarada)
367 EqGain, // +64 igual que el motor
368 ArpRes, // 0..5 real → índice del enum del motor (1/24 … 1/4)
369 ArpRange, // 0..3 real (1..4 octavas) → 1..4 del motor
370 FormantShift, // 0..4 real (0,+1,+2,-1,-2) → 0..4 del motor (-2,-1,0,+1,+2)
371 Duration127, // bits 0..6 (portamento)
372 SeqLastStep, // bits 4..7 almacenan último paso − 1 → 1..16 del motor
373 SeqMotionInv // bit 0 con el sentido invertido (real 0 = Smooth, motor 0 = Step)
374 };

378 size_t offset; // offset absoluto en el programa (ya sumado el timbre)

380 uint8_t bitHi; // inclusivo; bitHi < bitLo = byte completo

382 Xform xform;

385 static uint8_t extractBits(uint8_t byte, uint8_t lo, uint8_t hi)
386 {
387 if (hi < lo) return byte;
388 const uint8_t mask = static_cast<uint8_t>(((1u << (hi - lo + 1)) - 1u) << lo);
389 return static_cast<uint8_t>((byte & mask) >> lo);
390 }
391
392 static void insertBits(uint8_t& byte, uint8_t lo, uint8_t hi, uint8_t value)
393 {
394 if (hi < lo) { byte = value; return; }
395 const uint8_t mask = static_cast<uint8_t>(((1u << (hi - lo + 1)) - 1u) << lo);
396 byte = static_cast<uint8_t>((byte & ~mask) | ((value << lo) & mask));
397 }

399 /**
400 * Solo los campos que comparten byte con otros usan rango de bits; el resto
401 * ocupa el byte entero (`bitLo`/`bitHi` quedan ahí como documentación).
402 */
403 static bool usesBitRange(Xform xform)
404 {
405 // ArpRange comparte byte con el tipo de arpegio (bits 4..7), así que
406 // siempre se lee por rango; los demás ocupan el byte entero.
407 return xform == Xform::Bit || xform == Xform::Duration127 || xform == Xform::ArpRange
408 || xform == Xform::SeqLastStep || xform == Xform::SeqMotionInv;
409 }

411 /** `base` = 0 para el Timbre 1, `TIMBRE_SIZE` para el Timbre 2. */
412 uint8_t readField(const Field& f, size_t base = 0) const
413 {
414 const uint8_t byte = raw[f.offset + base];
415 return usesBitRange(f.xform) ? extractBits(byte, f.bitLo, f.bitHi) : byte;
416 }

418 void writeField(const Field& f, uint8_t value, size_t base = 0)
419 {
420 uint8_t& byte = raw[f.offset + base];
421 if (usesBitRange(f.xform)) insertBits(byte, f.bitLo, f.bitHi, value);
422 else byte = value;
423 }

612 static float toEngineValue(uint8_t fieldValue, Xform xform)
613 {
614 switch (xform)
615 {
616 case Xform::Signed: return static_cast<float>(static_cast<int>(fieldValue) - 64);
617 case Xform::EqFreq: return std::min(3.0f, std::floor(static_cast<float>(fieldValue) * 4.0f / 30.0f));
618 case Xform::EqGain: return static_cast<float>(fieldValue);
619 case Xform::ArpRes:
620 {
621 // 0..5 real = 1/24, 1/16, 1/12, 1/8, 1/6, 1/4 → índices del motor
622 static const float kTable[6] = { 2.0f, 3.0f, 4.0f, 5.0f, 6.0f, 7.0f };
623 return kTable[std::min<uint8_t>(fieldValue, 5)];
624 }
625 case Xform::FormantShift:
626 {
627 // real 0..4 = 0, +1, +2, -1, -2 → motor -2..+2
628 static const float kTable[5] = { 2.0f, 3.0f, 4.0f, 1.0f, 0.0f };
629 return kTable[std::min<uint8_t>(fieldValue, 4)];
630 }
631 case Xform::ArpRange: return static_cast<float>(fieldValue) + 1.0f;
632 case Xform::SeqLastStep: return static_cast<float>(fieldValue) + 1.0f;
633 case Xform::SeqMotionInv:
634 // El byte real guarda 0 = Smooth, 1 = Step; el motor numera al revés.
635 return (fieldValue == 0) ? static_cast<float>(ModSeqMotion::Smooth)
636 : static_cast<float>(ModSeqMotion::Step);
637 case Xform::Duration127:
638 case Xform::Raw:
639 case Xform::Bit:
640 default:
641 return static_cast<float>(fieldValue);
642 }
643 }
644
645 static uint8_t fromEngineValue(float value, Xform xform)
646 {
647 const int iv = static_cast<int>(std::lround(value));
648 switch (xform)
649 {
650 case Xform::Signed: return static_cast<uint8_t>(std::clamp(iv + 64, 0, 255));
651 case Xform::EqFreq:
652 {
653 // inversa: 4 pasos del motor → centro de cada tercio de los 30 pasos reales
654 static const int kTable[4] = { 3, 11, 18, 26 };
655 return static_cast<uint8_t>(kTable[std::clamp(iv, 0, 3)]);
656 }
657 case Xform::EqGain: return static_cast<uint8_t>(std::clamp(iv, 0, 127));
658 case Xform::ArpRes:
659 {
660 static const int kTable[16] = { -1, -1, 0, 1, 2, 3, 4, 5, -1, -1, -1, -1, -1, -1, -1, -1 };
661 const int mapped = kTable[std::clamp(iv, 0, 15)];
662 return static_cast<uint8_t>(mapped < 0 ? 3 : mapped); // fuera de la tabla: 1/16
663 }
664 case Xform::FormantShift:
665 {
666 static const int kTable[5] = { 3, 4, 0, 1, 2 }; // -2,-1,0,+1,+2 → real
667 return static_cast<uint8_t>(kTable[std::clamp(iv, 0, 4)]);
668 }
669 case Xform::ArpRange: return static_cast<uint8_t>(std::clamp(iv - 1, 0, 3));
670 case Xform::SeqLastStep: return static_cast<uint8_t>(std::clamp(iv - 1, 0, 15));
671 case Xform::SeqMotionInv:
672 return static_cast<uint8_t>(iv == static_cast<int>(ModSeqMotion::Smooth) ? 0 : 1);
673 case Xform::Duration127:
674 case Xform::Raw:
675 case Xform::Bit:
676 default:
677 return static_cast<uint8_t>(std::clamp(iv, 0, 255));
678 }
679 }

691 /**

693 * sequences. Devuelve el nº de parámetros escritos.
694 */
695 int applyToAPVTS(juce::AudioProcessorValueTreeState& apvts) const

698 auto setValueById = [&](const juce::String& id, float value)
699 {
700 if (auto* p = apvts.getParameter(id))
701 {
702 p->setValueNotifyingHost(p->convertTo0to1(value));
703 ++applied;

705 };

707 {
708 if (id == nullptr) return;
709 setValueById(timbreParamId(id, second), value);
710 };
711
712 const bool vocoder = isVocoderProgram();

714
715 // Modo de programa (0x10 bits 4,5) y reparto de las 4 voces (bits 6,7).
716 const uint8_t voiceModeBits = extractBits(raw[GLOBAL_FLAGS], 4, 5);
717 setValueById(ParamIDs::timbreMode, vocoder ? 0.0f : static_cast<float>(std::min<uint8_t>(voiceModeBits, 2)));
718 setValueById(ParamIDs::timbreVoices, static_cast<float>(extractBits(raw[GLOBAL_FLAGS], 6, 7)));
719
720 if (vocoder)
721 {
722 for (const auto& f : vocoderFields())
723 setValue(f.paramId, toEngineValue(readField(f), f.xform), false);
724 forEachVocoderBand([&](const char* id, size_t offset) { setValue(id, static_cast<float>(raw[offset]), false); });
725 }
726 else
727 {
728 // Los campos globales (FX, EQ, arpegiador, escala, split) una vez; los del
729 // bloque de timbre, dos veces: Timbre 1 con base 0 y Timbre 2 con base 108.
730 for (const auto& f : synthFields())
731 {
732 if (!isTimbreField(f))
733 {
734 setValue(f.paramId, toEngineValue(readField(f), f.xform), false);
735 continue;
736 }
737 for (size_t t = 0; t < 2; ++t)
738 setValue(f.paramId, toEngineValue(readField(f, t * TIMBRE_SIZE), f.xform), t == 1);
739 }
740 for (size_t t = 0; t < 2; ++t)
741 forEachSeqValue([&](const juce::String& id, float value) { setValueById(id, value); },
742 t == 1, t * TIMBRE_SIZE);
743 }
744
745 // El portamento del MS2000 no tiene interruptor: está activo cuando el tiempo > 0.
746 const uint8_t porta = extractBits(raw[TIMBRE1_START + ti::PORTAMENTO], 0, 6); // bit 7 sin usar
747 setValueById(ParamIDs::portamentoOn, porta > 0 ? 1.0f : 0.0f);
748
749 return applied;
750 }

752 /**
753 * Vuelca al programa (`raw`) todo lo que el motor modela: globales, **los dos timbres**
754 * y sus mod sequences. Lo que no modela (MIDI ch por timbre, transpose, bend, vibrato,
755 * resets de EG, prioridad… ) se queda como estaba en `raw`, así que un parche que vino
756 * del equipo vuelve al equipo sin perderlo.
757 *
758 * Es el **único** camino de escritura, y `getValueById` decide de dónde sale cada valor:
759 * del motor (`captureFromAPVTS`) o del bloque nativo del plugin (`fromNativeProgram`,
760 * §6.4 de `DOCS/ABDSynths_SysEx_Spec.md`). Así los dos sentidos mapean exactamente igual
761 * y no pueden desincronizarse.
762 */
763 template <typename GetValue>
764 int captureModeledFields(GetValue&& getValueById, const std::string& programName)
765 {
766 int captured = 0;
767 auto getValue = [&](const char* id, float fallback, bool second) -> float
768 {
769 if (id == nullptr) return fallback;
770 return getValueById(timbreParamId(id, second), fallback);
771 };
772 auto store = [&](const Field& f, float value, size_t base)
773 {
774 writeField(f, fromEngineValue(value, f.xform), base);
775 ++captured;
776 };
777
778 if (!programName.empty()) setName(programName);
779
780 const bool vocoder = getValueById(ParamIDs::synthVocoderMode, 0.0f) > 0.5f;
781 if (vocoder)
782 setVoiceMode(VoiceMode::Vocoder);
783 else
784 setVoiceMode(static_cast<VoiceMode>(std::max(0.0f, std::min(2.0f,
785 getValueById(ParamIDs::timbreMode, 0.0f)))));
786
787 if (vocoder)
788 {
789 for (const auto& f : vocoderFields())
790 {
791 if (f.paramId == nullptr) continue;
792 store(f, getValue(f.paramId, 0.0f, false), 0);
793 }
794 forEachVocoderBand([&](const char* id, size_t offset)
795 {
796 raw[offset] = static_cast<uint8_t>(std::clamp(static_cast<int>(std::lround(getValueById(id, 0.0f))), 0, 127));
797 });
798 }
799 else
800 {
801 for (const auto& f : synthFields())
802 {
803 if (f.paramId == nullptr) continue; // el voice mode global se trata aparte
804 if (!isTimbreField(f))
805 {
806 store(f, getValue(f.paramId, 0.0f, false), 0);
807 continue;
808 }
809 for (size_t t = 0; t < 2; ++t)
810 {
811 const bool second = (t == 1);
812 store(f, getValue(f.paramId, 0.0f, second), t * TIMBRE_SIZE);
813 }
814 }
815 for (size_t t = 0; t < 2; ++t)
816 captureSeqFromAPVTS(getValueById, t == 1, t * TIMBRE_SIZE);
817 }
818
819 const uint8_t porta = static_cast<uint8_t>(std::clamp(
820 static_cast<int>(std::lround(getValueById(ParamIDs::portamentoTime, 0.0f))), 0, 127));
821 insertBits(raw[TIMBRE1_START + ti::PORTAMENTO], 0, 6, porta);
822
823 return captured;
824 }

830 {

837 /**
838 * Valor del motor tal como está guardado en el **bloque nativo** del plugin
839 * (`MS2000ProgramData`, 384 B): un byte por parámetro en su `sysexOffset`, con los
840 * mismos sesgos que `MS2000ProgramData::applyToAPVTS` (booleano 64/0, con signo +64).

843 * propio offset, así que se leen aparte (igual que hace el propio bloque nativo).
844 */
845 static float nativeEngineValue(const MS2000ProgramData& native, const juce::String& id, float fallback)
846 {
847 const uint8_t voiceByte = native.rawData[MS2000ProgramData::VOICE_BYTE];
848 const char* raw = id.toRawUTF8();
849
850 if (std::strcmp(raw, ParamIDs::voiceMode) == 0)
851 return static_cast<float>(voiceByte & 0x03);
852 if (std::strcmp(raw, ParamIDs::unisonDetune) == 0)
853 return static_cast<float>((voiceByte >> 2) & 0x0F);
854 if (std::strcmp(raw, ParamIDs::portamentoOn) == 0)
855 return static_cast<float>((voiceByte >> 6) & 0x01);
856 if (std::strcmp(raw, ParamIDs::portamentoTime) == 0)
857 return static_cast<float>(native.rawData[MS2000ProgramData::VOICE_BYTE + 2]);
858
859 const auto* meta = ParameterRegistry::getParameter(raw);
860 if (meta == nullptr || meta->sysexOffset < 0) return fallback;
861
862 const size_t off = MS2000ProgramData::TIMBRE_START + static_cast<size_t>(meta->sysexOffset);
863 if (off >= MS2000ProgramData::UNPACKED_PROGRAM_SIZE) return fallback;
864
865 const uint8_t byte = native.rawData[off];
866 if (meta->type == ParamType::Boolean) return byte >= 64 ? 1.0f : 0.0f;
867 if (meta->min < 0.0f) return static_cast<float>(static_cast<int>(byte) - 64);
868 return static_cast<float>(byte);
869 }

871 /**
872 * Programa **real** del equipo construido desde un preset **nativo** del plugin
873 * (`MS2000ProgramData`): el paso que permite que el plugin suelto haga de equipo y
874 * conteste un `0x0E` de Korg con su memoria completa (§6.4).
875 *
876 * ⚠️ Conversión **declarada como aproximada**: parte de la plantilla "INIT Program" del
877 * equipo y escribe encima todo lo que el motor modela, así que los campos que el motor
878 * no modela (`unmodelled()`: MIDI ch por timbre, Tune, Bend Range, Transpose, Vibrato,
879 * resets de EG…) salen con los valores del INIT — no se inventa nada, pero tampoco se
880 * puede recuperar lo que el bloque nativo nunca guardó. **Pendiente de verificar contra
881 * un MS2000 físico** (ver el roadmap de ABDMS2000).
882 */
883 static MS2000HardwareProgram fromNativeProgram(const MS2000ProgramData& native,
884 const std::string& programName = {})
885 {

887 auto getValueById = [&native](const juce::String& id, float fallback) -> float
888 {
889 return nativeEngineValue(native, id, fallback);
890 };
891 program.captureModeledFields(getValueById, programName.empty() ? native.getName() : programName);
892 return program;
893 }
