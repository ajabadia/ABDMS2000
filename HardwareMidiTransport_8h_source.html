10/**
11 * Transporte MIDI de hardware del anfitrión: el otro extremo del puente MIDI del
12 * Bank Manager embebido.
13 *
14 * Dentro del WebView2 no hay Web MIDI, así que el Bank Manager construye el SysEx
15 * del contrato y lo manda por el bridge (`hardware.send`); este transporte lo
16 * entrega al dispositivo MIDI y devuelve a la WebUI los bytes que lleguen
17 * (`hardware.receive`). `hardware.listen` es el Bank Manager diciendo que ya está
18 * suscrito, momento en el que merece la pena abrir la entrada.
19 *
20 * Los dispositivos los abre el Editor (`bind()`), que sí puede enlazar
21 * `juce_audio_devices`; este header solo depende de `juce_core` para poder
22 * compilarse también en `ABDMS2000_Tests`, que no enlaza ese módulo.
23 *
24 * Sin `bind()` (tests, o un host sin hardware disponible) todo falla con un motivo
25 * explícito en lugar de tragarse los bytes en silencio: así el Bank Manager puede
26 * decir *por qué* no habrá fetch en vez de dejar al usuario esperando un timeout.
27 * (Antes de existir este transporte, `hardware.send` / `hardware.listen` no se
28 * despachaban en absoluto y el fetch por el puente solo podía morir por timeout.)
29 */
30class HardwareMidiTransport
31{
32public:
33 /** Resultado de una operación: `detail` es el dispositivo (si ok) o el motivo. */
34 struct Outcome
35 {
36 bool ok = false;
37 juce::String detail;
38 };
39
40 /** WebUI → hardware. Devuelve ok=false con el motivo cuando no se pudo enviar. */
41 using SendFunction = std::function<Outcome(const juce::MemoryBlock&)>;
42 /** El Bank Manager ya escucha: abre la entrada. `detail` = nombre del dispositivo. */
43 using ListenFunction = std::function<Outcome()>;
44 /** Enumeración de puertos para `hardware.listPorts`; devuelve `{ inputs, outputs }`. */
45 using PortListFunction = std::function<juce::var()>;
46 /** Envoltorio JS ya formado (`{ type, data }`) para los bytes que llegan. */
47 using ReceiveSink = std::function<void(const juce::var&)>;
48
49 /** Inyecta el hardware. Sin argumentos válidos, `send`/`listen` fallan con motivo. */
50 void bind(SendFunction send, ListenFunction listen, ReceiveSink sink)
51 {
52 sendFunction = std::move(send);
53 listenFunction = std::move(listen);
54 receiveSink = std::move(sink);
55 listening = false;
56 lastDetail.clear();
57 }
58
59 /** Instala la enumeración que solo conoce el Editor JUCE. */
60 void setPortListFunction(PortListFunction fn) { portListFunction = std::move(fn); }
61
62 /** Lista los puertos actuales; una ausencia de proveedor devuelve un objeto vacío. */
63 juce::var listPorts() const
64 {
65 if (portListFunction)
66 return portListFunction();
67 return juce::var(new juce::DynamicObject());
68 }
69
70 /** Selección explícita del Bank Manager. No se elige ningún primer puerto implícitamente. */
71 void selectPorts(const juce::String& outputId, const juce::String& inputId)
72 {
73 selectedOutputId = outputId;
74 selectedInputId = inputId;
75 listening = false;
76 lastDetail.clear();
77 }
78
79 const juce::String& getSelectedOutputId() const noexcept { return selectedOutputId; }
80 const juce::String& getSelectedInputId() const noexcept { return selectedInputId; }
81 bool hasSelectedOutput() const noexcept { return selectedOutputId.isNotEmpty(); }
82 bool hasSelectedInput() const noexcept { return selectedInputId.isNotEmpty(); }
83
84 /** La entrada seleccionada desapareció: no se deben reenviar datos obsoletos. */
85 void markInputUnavailable(const juce::String& reason)
86 {
87 listening = false;
88 lastDetail = reason;
89 }
90
91 /** El Editor se va (o se quedan sin dispositivo): deja de haber hardware. */
92 void unbind()
93 {
94 listening = false;
95 sendFunction = {};
96 listenFunction = {};
97 portListFunction = {};
98 receiveSink = {};

100 }
101
102 bool isBound() const noexcept { return static_cast<bool>(sendFunction); }
103 bool isListening() const noexcept { return listening; }
104 const juce::String& getLastDetail() const noexcept { return lastDetail; }
105
106 /** WebUI → hardware. */
107 bool send(const juce::MemoryBlock& message)
108 {
109 if (message.isEmpty())
110 return fail("Empty MIDI message");
111
112 if (! sendFunction)
113 return fail("No MIDI output device available for the hardware transfer");
114
115 auto outcome = sendFunction(message);
116 lastDetail = outcome.detail;
117 return outcome.ok;
118 }
119
120 /** El Bank Manager ha pedido escuchar: arranca la entrada y reenvía lo que llegue. */
121 bool listen()
122 {
123 if (! listenFunction)
124 return fail("No MIDI input device available for the hardware transfer");
125
126 auto outcome = listenFunction();
127 listening = outcome.ok;
128 lastDetail = outcome.detail;
129 return outcome.ok;
130 }
131
132 /**
133 * Bytes del hardware → WebUI, con la misma forma que el core del Bank Manager
134 * standalone: `{ type: 'hardware.receive', data: { payload: <base64>, size } }`.
135 */
136 void deliverIncoming(const juce::MemoryBlock& message)
137 {
138 if (! listening || message.isEmpty() || ! receiveSink)
139 return;
140
141 juce::DynamicObject::Ptr data = new juce::DynamicObject();
142 data->setProperty("payload", juce::Base64::toBase64(message.getData(), message.getSize()));
143 data->setProperty("size", static_cast<int>(message.getSize()));
144
145 juce::DynamicObject::Ptr envelope = new juce::DynamicObject();
146 envelope->setProperty("type", "hardware.receive");
147 envelope->setProperty("data", juce::var(data.get()));
148
149 receiveSink(juce::var(envelope.get()));
150 }
151
152private:
153 bool fail(const juce::String& reason)
154 {
155 lastDetail = reason;
156 return false;
157 }
158
159 SendFunction sendFunction;
160 ListenFunction listenFunction;
161 PortListFunction portListFunction;
162 ReceiveSink receiveSink;
163 bool listening = false;
164 juce::String lastDetail;
165 juce::String selectedOutputId;
166 juce::String selectedInputId;
167};
