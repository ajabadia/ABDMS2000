17/**
18 * Transporte MIDI de hardware del anfitrión: el otro extremo del puente MIDI del
19 * Bank Manager embebido y de la utilidad de actualización de firmware IPL v1.07.
20 *
21 * Dentro del WebView2 no hay Web MIDI, así que el Bank Manager construye el SysEx
22 * del contrato y lo manda por el bridge (`hardware.send`); este transporte lo
23 * entrega al dispositivo MIDI y devuelve a la WebUI los bytes que lleguen
24 * (`hardware.receive`). `hardware.listen` es el Bank Manager diciendo que ya está
25 * suscrito, momento en el que merece la pena abrir la entrada.
26 *
27 * Los dispositivos los abre el Editor (`bind()`), que sí puede enlazar
28 * `juce_audio_devices`; este header solo depende de `juce_core` para poder
29 * compilarse también en `ABDMS2000_Tests`, que no enlaza ese módulo.
30 *
31 * Sin `bind()` (tests, o un host sin hardware disponible) todo falla con un motivo
32 * explícito en lugar de tragarse los bytes en silencio: así el Bank Manager puede
33 * decir *por qué* no habrá fetch en vez de dejar al usuario esperando un timeout.
34 * (Antes de existir este transporte, `hardware.send` / `hardware.listen` no se
35 * despachaban en absoluto y el fetch por el puente solo podía morir por timeout.)
36 */

166 /**
167 * Bytes del hardware → WebUI, con la misma forma que el core del Bank Manager
168 * standalone: `{ type: 'hardware.receive', data: { payload: <base64>, size } }`.
169 */
170 void deliverIncoming(const juce::MemoryBlock &message)
171 {
172 if (!listening || message.isEmpty() || !receiveSink)
173 return;
174
175 juce::DynamicObject::Ptr data = new juce::DynamicObject();
176 data->setProperty("payload", juce::Base64::toBase64(message.getData(), message.getSize()));
177 data->setProperty("size", static_cast<int>(message.getSize()));
178
179 juce::DynamicObject::Ptr envelope = new juce::DynamicObject();
180 envelope->setProperty("type", "hardware.receive");
181 envelope->setProperty("data", juce::var(data.get()));
182
183 receiveSink(juce::var(envelope.get()));
184 }

193 void deliverIncomingSysEx(const uint8_t *payloadWithoutF0F7, std::size_t payloadSize)
194 {
195 if (payloadWithoutF0F7 == nullptr)
196 return;
197
198 std::vector<uint8_t> full;
199 full.reserve(payloadSize + 2);
200 full.push_back(0xF0);
201 full.insert(full.end(), payloadWithoutF0F7, payloadWithoutF0F7 + payloadSize);
202 full.push_back(0xF7);
203
204 {
205 std::lock_guard<std::mutex> lock(sysexMutex_);
206 if (sysexQueue_.size() < 64)
207 sysexQueue_.push_back(std::move(full));
208 }
209 sysexCv_.notify_one();
210 }
211
212 bool waitForSysEx(std::vector<uint8_t> &outSysEx, int timeoutMs)
213 {
214 std::unique_lock<std::mutex> lock(sysexMutex_);
215 if (!sysexCv_.wait_for(lock, std::chrono::milliseconds(timeoutMs), [this]() { return !sysexQueue_.empty(); }))
216 return false;
217
218 outSysEx = std::move(sysexQueue_.front());
219 sysexQueue_.pop_front();
220 return true;
221 }
