22 void drawRotarySlider(juce::Graphics& g, int x, int y, int width, int height,
23 float sliderPosProportional, float rotaryStartAngle, float rotaryEndAngle,
24 juce::Slider& slider) override
25 {
26 // 1. Calcular el área y radio del potenciómetro de forma proporcional
27 auto radius = static_cast<float>(juce::jmin(width / 2, height / 2)) - 4.0f;
28 auto centreX = static_cast<float>(x) + static_cast<float>(width) * 0.5f;
29 auto centreY = static_cast<float>(y) + static_cast<float>(height) * 0.5f;
30 auto rx = centreX - radius;
31 auto ry = centreY - radius;
32 auto rw = radius * 2.0f;
33
34 // 2. Dibujar la sombra exterior proyectada sobre el chasis (Look analógico)
35 g.setColour(juce::Colours::black.withAlpha(0.25f));
36 g.fillEllipse(rx + 1.0f, ry + 2.0f, rw, rw);
37
38 // 3. Dibujar la falda/base del potenciómetro (Plástico estriado exterior)
39 juce::Colour bodyColor = slider.findColour(juce::Slider::rotarySliderFillColourId);
40 g.setColour(bodyColor.darker(0.15f));
41 g.fillEllipse(rx, ry, rw, rw);
42
43 // 4. Dibujar la tapa superior (Metal/Plástico cóncavo con sutil gradiente)
44 float topRadiusOffset = radius * 0.12f;
45 auto trx = rx + topRadiusOffset;
46 auto tryY = ry + topRadiusOffset;
47 auto trw = rw - (topRadiusOffset * 2.0f);
48
49 juce::ColourGradient capGradient(bodyColor.brighter(0.08f), centreX, centreY - radius,
50 bodyColor.darker(0.2f), centreX, centreY + radius, false);
51 g.setBrush(capGradient);
52 g.fillEllipse(trx, tryY, trw, trw);
53
54 // 5. Dibujar el borde metálico circular exterior
55 g.setColour(juce::Colours::black.withAlpha(0.4f));
56 g.drawEllipse(trx, tryY, trw, trw, 1.0f);
57
58 // 6. Cálculo de la posición y dibujo de la línea indicadora de Korg
59 auto angle = rotaryStartAngle + sliderPosProportional * (rotaryEndAngle - rotaryStartAngle);
60
61 float pointerLength = trw * 0.42f;
62 float pointerThickness = 2.0f;
63
64 auto pointerStartX = centreX + std::sin(angle) * (trw * 0.1f);
65 auto pointerStartY = centreY - std::cos(angle) * (trw * 0.1f);
66 auto pointerEndX = centreX + std::sin(angle) * pointerLength;
67 auto pointerEndY = centreY - std::cos(angle) * pointerLength;
68
69 g.setColour(slider.findColour(juce::Slider::thumbColourId));
70 g.drawLine(pointerStartX, pointerStartY, pointerEndX, pointerEndY, pointerThickness);
71
72 // Sutil punto central
73 g.setColour(bodyColor.darker(0.5f));
74 g.fillEllipse(centreX - 1.5f, centreY - 1.5f, 3.0f, 3.0f);
75 }

77 void drawButtonBackground(juce::Graphics& g, juce::Button& button,
78 const juce::Colour& backgroundColour,
79 bool shouldDrawButtonAsHighlighted,
80 bool shouldDrawButtonAsDown) override
81 {
82 auto bounds = button.getLocalBounds().toFloat();
83 auto cornerSize = 2.5f;
84
85 juce::Colour baseColor = backgroundColour;
86 if (shouldDrawButtonAsHighlighted) baseColor = baseColor.brighter(0.08f);
87 if (shouldDrawButtonAsDown) baseColor = baseColor.darker(0.15f);
88
89 // 1. Sombra proyectada
90 g.setColour(juce::Colours::black.withAlpha(0.3f));
91 g.fillRoundedRectangle(bounds.translated(0.0f, 1.5f).reduced(0.5f), cornerSize);
92
93 // 2. Cuerpo del botón
94 juce::ColourGradient bodyGrad(baseColor.brighter(0.12f), bounds.getX(), bounds.getY(),
95 baseColor.darker(0.22f), bounds.getX(), bounds.getBottom(), false);
96 g.setBrush(bodyGrad);
97 g.fillRoundedRectangle(bounds.reduced(1.0f), cornerSize);
98
99 // 3. Bisel interior
100 g.setColour(baseColor.brighter(0.25f).withAlpha(0.4f));
101 g.drawRoundedRectangle(bounds.reduced(1.5f), cornerSize, 1.0f);
102
103 // 4. Borde exterior
104 g.setColour(juce::Colours::black.withAlpha(0.6f));
105 g.drawRoundedRectangle(bounds.reduced(1.0f), cornerSize, 1.0f);
106
107 // 5. Diodo LED integrado
108 if (button.getToggleState() || shouldDrawButtonAsDown)
109 {
110 float ledSize = 4.5f;
111 float ledX = (bounds.getWidth() - ledSize) * 0.5f;
112 float ledY = 5.0f;
113
114 juce::Colour ledRed = juce::Colour::fromRGBA(255, 40, 20, 255);
115 g.setColour(ledRed);
116 g.fillEllipse(ledX, ledY, ledSize, ledSize);
117
118 // Glow analógico
119 g.setColour(ledRed.withAlpha(0.4f));
120 g.fillEllipse(ledX - 2.0f, ledY - 2.0f, ledSize + 4.0f, ledSize + 4.0f);
121
122 g.setColour(ledRed.withAlpha(0.15f));
123 g.fillEllipse(ledX - 4.0f, ledY - 4.0f, ledSize + 8.0f, ledSize + 8.0f);
124 }
125 else
126 {
127 float ledSize = 4.5f;
128 float ledX = (bounds.getWidth() - ledSize) * 0.5f;
129 float ledY = 5.0f;
130
131 g.setColour(juce::Colour::fromRGBA(80, 15, 10, 255));
132 g.fillEllipse(ledX, ledY, ledSize, ledSize);
133
134 g.setColour(juce::Colours::black.withAlpha(0.3f));
135 g.drawEllipse(ledX, ledY, ledSize, ledSize, 0.5f);
