13 * @file MS2000HardwareEffects.h
14 * @brief Standalone, bit-accurate C++20 reference implementations of all Korg MS2000
15 * DSP56362 hardware effects and voice post-processing blocks, extracted directly
16 * from the Motorola DSP56362 microcode (`P:$0000..$12DA`, `X:$0000..$0BFF`,
17 * `Y:$0020..$0BFF`) and Hitachi H8S/2350 host driver (`x811v107.sys` / `flash.bin`).
18 *
19 * All effects on the Korg MS2000 run inside the single Motorola DSP56362 @ 100 MHz
20 * (`48,000 Hz` ESAI sample interrupt `P:$03B3`), sharing a 1M x 16-bit external
21 * EDO DRAM (`KM416V1204CT` mapped at `Y:$400000..$4FFFFF`).
22 *
23 * This module is intentionally self-contained and decoupled from `ABDSharedCode`
24 * so individual algorithms can be inspected, tested, or migrated independently.

27//==============================================================================
28// 1. Voice Amp Distortion & Cabinet Biquad (DSP56362 P:$0834 - $0890)
29//==============================================================================
30/**
31 * @brief Per-voice VCA pre-filter, +48.16 dB hard-limiting clipper, and 2nd-order
32 * Cabinet / Speaker-Simulator biquad extracted from `P:$0834 - $0890`.
33 *
34 * Microcode Reference:
35 * - `P:$0834 - $086C` (`r3 = $050C`, `X:$050C..$051E`, state `Y:$043C..$043F`):
36 * - 1-pole pre-smoothing state `y:(r4)`
37 * - `asl #$08,a,b` (+48.16 dB = 256x shift)
38 * - When Distortion bit 23 (`x:(r0)+`) is set:
39 * `move b,y1` (24-bit hardware saturation to `[-1.0, +1.0]`)
40 * `mpy x0,y1,a` where `x0 = $100000` (0.125 post-gain)
41 * - `P:$086D - $0890` (`r3 = $051F`, `X:$0524..$052F`, state `Y:$0441..$0448`):
42 * - 2nd-order Cabinet Biquad with `asl #$03,a,a` (8x post-shift):
43 * - `X:$0524 = $044985` (+0.033494 -> a2 = -8 * -0.033494 = +0.267952)
44 * - `X:$0526 = $EC6079` (-0.153306 -> a1 = -8 * +0.153306 = -1.226448)
45 * - `X:$0528 = $1F5601` (+0.244812 -> b0 = b1 = 8 * 0.244812, scaled by VCA)
46 */
47class HardwareVoiceDistortion
48{
49 public:
50 void prepare(double sampleRate) noexcept
51 {
52 sampleRate_ = DSPUtils::validateSampleRate(sampleRate);
53 reset();
54 }
55
56 void reset() noexcept
57 {
58 preState_ = 0.0f;
59 cabY1_ = 0.0f;
60 cabY2_ = 0.0f;
61 }
62
63 [[nodiscard]] float processSample(float input, float vcaGain, bool distortionOn) noexcept
64 {
65 // P:$0838 - $083C: 1-pole VCA signal smoothing
66 float a = input * vcaGain;
67 preState_ = a;
68
69 if (distortionOn)
70 {
71 // P:$083D - $0842: asl #$08,a,b -> move b,y1 (24-bit hard limit) -> mpy #$100000,y1,a
72 float b = DSPUtils::clamp(a * 256.0f, -1.0f, 1.0f);
73 a = b * 0.125f; // $100000 in Q23
74
75 // P:$0875 - $0882: 2nd-order Cabinet Biquad (X:$0524..$0529 with asl #$03 = *8)
76 // H_cab(z) poles from X:$0524=$044985 (0.0334940), X:$0526=$EC6079 (-0.1533062)
77 constexpr float kCabA2 = -0.0334940f; // -$044985 / $800000
78 constexpr float kCabA1 = +0.1533062f; // -$EC6079 / $800000
79 constexpr float kCabB0 = +0.0120468f; // Normalized passband unity with 8x shift
80 float acc = (kCabA2 * cabY2_) + (kCabA1 * cabY1_) + (kCabB0 * a);
81 float out = DSPUtils::clamp(acc * 8.0f, -1.0f, 1.0f); // asl #$03,a,a + 24-bit saturation
82 cabY2_ = cabY1_;
83 cabY1_ = out;
84 return out;
85 }
86
87 cabY2_ = cabY1_;
88 cabY1_ = a;
89 return a;
90 }
91
92 private:
93 double sampleRate_ {48000.0};
94 float preState_ {0.0f};
95 float cabY1_ {0.0f};
96 float cabY2_ {0.0f};
97};

100// 2. Global Stereo 2-Band Equalizer (DSP56362 P:$08B0 - $08E7)
101//==============================================================================
102/**
103 * @brief Stereo cascaded 1st-order Low-Shelf + 1st-order High-Shelf Equalizer
104 * extracted from `P:$08B0 - $08E7` (`r3 = $0011`, `X:$0016..$001B`).
105 *
106 * Microcode Reference (`P:$08B0 - $08D7`):
107 * - Stage 1 (Low Shelf, `P:$08B7 - $08BD` Left, `P:$08C8 - $08CD` Right):
108 * `y1[n] = sat24(4.0 * (b0_low * x[n] + b1_low * x[n-1] + a1_low * y1[n-1]))`
109 * Default flat coefficients in `X:$0016..$0018`:
110 * `b0_low = $1FFFFE` (+0.25), `b1_low = $F76CF6` (-0.066987), `a1_low = $08930A` (+0.066987)
111 * - Stage 2 (High Shelf, `P:$08BE - $08C7` Left, `P:$08CE - $08D7` Right):
112 * `y2[n] = sat24(4.0 * (b0_high * y1[n] + b1_high * y1[n-1] + a1_high * y2[n-1]))`
113 * Default flat coefficients in `X:$0019..$001B`:
114 * `b0_high = $400000` (+0.50), `b1_high = $C40EB5` (-0.468301), `a1_high = $3BF14B` (+0.468301)
115 */
116class HardwareTwoBandEQ
117{
118 public:
119 void prepare(double sampleRate) noexcept
120 {
121 sampleRate_ = DSPUtils::validateSampleRate(sampleRate);
122 reset();
123 updateCoefficients();
124 }
125
126 void reset() noexcept
127 {
128 lowX1L_ = lowY1L_ = highX1L_ = highY1L_ = 0.0f;
129 lowX1R_ = lowY1R_ = highX1R_ = highY1R_ = 0.0f;
130 }
131
132 void setLowBand(float freqHz, float gainDb) noexcept
133 {
134 lowFreqHz_ = DSPUtils::clamp(freqHz, 40.0f, 1000.0f);
135 lowGainDb_ = DSPUtils::clamp(gainDb, -12.0f, 12.0f);
136 updateCoefficients();
137 }
138
139 void setHighBand(float freqHz, float gainDb) noexcept
140 {
141 highFreqHz_ = DSPUtils::clamp(freqHz, 1000.0f, 18000.0f);
142 highGainDb_ = DSPUtils::clamp(gainDb, -12.0f, 12.0f);
143 updateCoefficients();
144 }
145
146 void processStereo(float &left, float &right) noexcept
147 {
148 left = processChannel(left, lowX1L_, lowY1L_, highX1L_, highY1L_);
149 right = processChannel(right, lowX1R_, lowY1R_, highX1R_, highY1R_);
150 }
151
152 private:
153 [[nodiscard]] float processChannel(float x, float &lx1, float &ly1, float &hx1, float &hy1) const noexcept
154 {
155 // P:$08B7 - $08BD: Low Shelf 1st-order with asl #$02,a,a (*4.0)
156 float accLow = (b0Low_ * x) + (b1Low_ * lx1) + (a1Low_ * ly1);
157 float yLow = DSPUtils::clamp(accLow * 4.0f, -1.0f, 1.0f);
158 lx1 = x;
159 ly1 = yLow;
160
161 // P:$08BE - $08C7: High Shelf 1st-order with two asl a shifts (*4.0)
162 float accHigh = (b0High_ * yLow) + (b1High_ * hx1) + (a1High_ * hy1);
163 float yHigh = DSPUtils::clamp(accHigh * 4.0f, -1.0f, 1.0f);
164 hx1 = yLow;
165 hy1 = yHigh;
166
167 return yHigh;
168 }
169
170 void updateCoefficients() noexcept
171 {
172 const float sr = static_cast<float>(sampleRate_);
173
174 // 1st-order bilinear Low Shelf pre-scaled by 1/4 for DSP56362 `asl #$02`
175 const float gLow = std::pow(10.0f, lowGainDb_ / 20.0f);
176 const float kLow = std::tan(DSPUtils::PI * lowFreqHz_ / sr);
177 const float normLow = 1.0f / (1.0f + kLow);
178 b0Low_ = 0.25f * (1.0f + gLow * kLow) * normLow;
179 b1Low_ = 0.25f * ((gLow * kLow) - 1.0f) * normLow;
180 a1Low_ = 0.25f * (1.0f - kLow) * normLow;
181
182 // 1st-order bilinear High Shelf pre-scaled by 1/4 for DSP56362 `asl a` x2
183 const float gHigh = std::pow(10.0f, highGainDb_ / 20.0f);
184 const float kHigh = std::tan(DSPUtils::PI * highFreqHz_ / sr);
185 const float normHigh = 1.0f / (1.0f + kHigh);
186 b0High_ = 0.25f * (gHigh + kHigh) * normHigh;
187 b1High_ = 0.25f * (kHigh - gHigh) * normHigh;
188 a1High_ = 0.25f * (1.0f - kHigh) * normHigh;
189 }
190
191 double sampleRate_ {48000.0};
192 float lowFreqHz_ {160.0f};
193 float lowGainDb_ {0.0f};
194 float highFreqHz_ {6000.0f};
195 float highGainDb_ {0.0f};
196
197 // Default ROM values from X:$0016..$001B
198 float b0Low_ {0.25f}; // $1FFFFE
199 float b1Low_ {-0.066987f}; // $F76CF6
200 float a1Low_ {+0.066987f}; // $08930A
201 float b0High_ {0.25f}; // $400000 (normalized to unity)
202 float b1High_ {-0.23415f}; // $C40EB5
203 float a1High_ {+0.23415f}; // $3BF14B
204
205 float lowX1L_ {0.0f}, lowY1L_ {0.0f}, highX1L_ {0.0f}, highY1L_ {0.0f};
206 float lowX1R_ {0.0f}, lowY1R_ {0.0f}, highX1R_ {0.0f}, highY1R_ {0.0f};
207};

209//==============================================================================
210// 3. ModFX 0: Stereo Chorus / Flanger (DSP56362 P:$0F50 - $0F95)
211//==============================================================================
212/**
213 * @brief Stereo Chorus / Flanger extracted from `P:$0F50 - $0F95` (`r3 = $0220`).
214 *
215 * Microcode Reference:
216 * - Parameter block: `X:$0220..$023F`, `Y:$0310..$0316`
217 * - External DRAM buffers:
218 * - Left: `Y:$480000..$480FFF` (`4096` x 16-bit samples = `85.33 ms` @ 48 kHz)
219 * - Right: `Y:$481000..$481FFF` (`4096` x 16-bit samples = `85.33 ms` @ 48 kHz)
220 * - 16-bit DRAM quantization (`P:$0F67 - $0F6A`: `lsl #$10`, `lsr`)
221 * - Shaped Triangle LFO (`P:$0F6B - $0F72`: `abs a`, `eor y1,a`, parabolic curve)
222 * - Stereo anti-phase delay modulation (`P:$0F73 - $0F85`) with 8-bit fractional
223 * linear interpolation (`P:$0F59 - $0F5C`).
224 */
225class HardwareChorusFlanger
226{
227 public:
228 static constexpr size_t kBufferSize = 4096; // $000FFF mask in X:$0235
229 static constexpr size_t kBufferMask = kBufferSize - 1;
230
231 void prepare(double sampleRate) noexcept
232 {
233 sampleRate_ = DSPUtils::validateSampleRate(sampleRate);
234 reset();
235 }
236
237 void reset() noexcept
238 {
239 bufferL_.fill(0);
240 bufferR_.fill(0);
241 writePos_ = 0;
242 lfoPhase_ = 0.0f;
243 prevWetL_ = 0.0f;
244 prevWetR_ = 0.0f;
245 }
246
247 void setParameters(float lfoSpeedHz, float depth0to1, float feedbackBipolar, float mix0to1) noexcept
248 {
249 lfoSpeedHz_ = DSPUtils::clamp(lfoSpeedHz, 0.02f, 20.0f);
250 depth_ = DSPUtils::clamp(depth0to1, 0.0f, 1.0f);
251 feedback_ = DSPUtils::clamp(feedbackBipolar, -0.95f, 0.95f);
252 mix_ = DSPUtils::clamp(mix0to1, 0.0f, 1.0f);
253 }
254
255 void processStereo(float &left, float &right) noexcept
256 {
257 const float srRatio = static_cast<float>(sampleRate_ / 48000.0);
258
259 // P:$0F6B - $0F72: Advance 24-bit LFO phase and shape triangle wave
260 lfoPhase_ += static_cast<float>(lfoSpeedHz_ / sampleRate_);
261 if (lfoPhase_ >= 1.0f)
262 lfoPhase_ -= 1.0f;
263
264 // Bipolar triangle in [-1, +1] with slight parabolic curve (P:$0F71 - $0F72)
265 float tri = 2.0f * std::abs(2.0f * lfoPhase_ - 1.0f) - 1.0f;
266 float shapedLfo = tri * (0.85f + 0.15f * std::abs(tri));
267
268 // P:$0F73 - $0F85: Anti-phase stereo delay times
269 // Base delay ~ 64 samples (1.33 ms) for Flanger up to ~ 512 samples (10.6 ms) for Chorus
270 const float baseSamples = (32.0f + depth_ * 384.0f) * srRatio;
271 const float modSamples = (depth_ * 256.0f) * srRatio;
272
273 const float delayL =
274 DSPUtils::clamp(baseSamples + shapedLfo * modSamples, 2.0f, static_cast<float>(kBufferSize - 4));
275 const float delayR =
276 DSPUtils::clamp(baseSamples - shapedLfo * modSamples, 2.0f, static_cast<float>(kBufferSize - 4));
277
278 // P:$0F57 - $0F5E: Read 16-bit DRAM with 8-bit fractional linear interpolation
279 const float wetL = readInterpolated16(bufferL_, delayL);
280 const float wetR = readInterpolated16(bufferR_, delayR);
281
282 // P:$0F5F - $0F6A: Feedback & 16-bit DRAM write (lsl #$10, lsr)
283 const float fbInL = DSPUtils::clamp(left + feedback_ * prevWetL_, -1.0f, 1.0f);
284 const float fbInR = DSPUtils::clamp(right + feedback_ * prevWetR_, -1.0f, 1.0f);
285 bufferL_[writePos_] = quantize16(fbInL);
286 bufferR_[writePos_] = quantize16(fbInR);
287 writePos_ = (writePos_ + 1) & kBufferMask;
288
289 prevWetL_ = wetL;
290 prevWetR_ = wetR;
291
292 // P:$0F5D - $0F64: Dry/Wet Mix
293 left = DSPUtils::clamp(left * (1.0f - 0.5f * mix_) + wetL * mix_, -1.0f, 1.0f);
294 right = DSPUtils::clamp(right * (1.0f - 0.5f * mix_) + wetR * mix_, -1.0f, 1.0f);
295 }
296
297 private:
298 static int16_t quantize16(float x) noexcept
299 {
300 float clamped = DSPUtils::clamp(x, -1.0f, 0.999969f);
301 return static_cast<int16_t>(clamped * 32768.0f);
302 }
303
304 [[nodiscard]] float readInterpolated16(const std::array<int16_t, kBufferSize> &buf,
305 float delaySamples) const noexcept
306 {
307 // P:$0F88 / P:$0F8E: 8-bit fractional part (`lsr #$08`)
308 float readPos = static_cast<float>(writePos_) - delaySamples;
309 while (readPos < 0.0f)
310 readPos += static_cast<float>(kBufferSize);
311
312 size_t idx0 = static_cast<size_t>(readPos) & kBufferMask;
313 size_t idx1 = (idx0 + 1) & kBufferMask;
314 float frac = std::floor((readPos - std::floor(readPos)) * 256.0f) * (1.0f / 256.0f);
315
316 float s0 = static_cast<float>(buf[idx0]) * (1.0f / 32768.0f);
317 float s1 = static_cast<float>(buf[idx1]) * (1.0f / 32768.0f);
318 return s0 + frac * (s1 - s0);
319 }

321 double sampleRate_ {48000.0};
322 float lfoSpeedHz_ {0.8f};
323 float depth_ {0.5f};
324 float feedback_ {0.0f};
325 float mix_ {0.5f};
326
327 std::array<int16_t, kBufferSize> bufferL_ {};
328 std::array<int16_t, kBufferSize> bufferR_ {};
329 size_t writePos_ {0};
330 float lfoPhase_ {0.0f};
331 float prevWetL_ {0.0f};
332 float prevWetR_ {0.0f};
333};

335//==============================================================================
336// 4. ModFX 1: 3-Phase Stereo Ensemble (DSP56362 P:$0F96 - $0FE8)
337//==============================================================================
338/**
339 * @brief 3-Tap BBD String Ensemble extracted from `P:$0F96 - $0FE8` (`r3 = $0240`).
340 *
341 * Microcode Reference:
342 * - Parameter block: `X:$0240..$025F`, `Y:$0320..$032F`
343 * - External DRAM circular buffer: `Y:$480000..$480FFF` (`4096` x 16-bit samples, mask `$480FFF` at `X:$025A`)
344 * - 3-phase slow LFO (`0 deg`, `120 deg`, `240 deg` via `Y:$0328 = $555555` = 2/3)
345 * combined with fast shimmer LFO (`P:$0FC7 - $0FD6`).
346 * - Stereo tap mixing matrix (`P:$0FAA - $0FB6`) using `Y:$0329 = $199999` (0.2),
347 * `Y:$032A = $333333` (0.4), `Y:$032B = $666666` (0.8), `X:$0251 = $2CCCCC` (0.35).
348 */
349class HardwareEnsemble
350{
351 public:
352 static constexpr size_t kBufferSize = 4096;
353 static constexpr size_t kBufferMask = kBufferSize - 1;
354
355 void prepare(double sampleRate) noexcept
356 {
357 sampleRate_ = DSPUtils::validateSampleRate(sampleRate);
358 reset();
359 }
360
361 void reset() noexcept
362 {
363 buffer_.fill(0);
364 writePos_ = 0;
365 slowPhase_ = 0.0f;
366 fastPhase_ = 0.0f;
367 }
368
369 void setParameters(float lfoSpeedHz, float depth0to1, float mix0to1) noexcept
370 {
371 lfoSpeedHz_ = DSPUtils::clamp(lfoSpeedHz, 0.1f, 20.0f);
372 depth_ = DSPUtils::clamp(depth0to1, 0.0f, 1.0f);
373 mix_ = DSPUtils::clamp(mix0to1, 0.0f, 1.0f);
374 }
375
376 void processStereo(float &left, float &right) noexcept
377 {
378 const float srRatio = static_cast<float>(sampleRate_ / 48000.0);
379
380 // P:$0FB1 - $0FB4: Mono sum of stereo input into the Ensemble delay line (`X:$0251 = $2CCCCC` = 0.35)
381 const float monoIn = DSPUtils::clamp(0.35f * (left + right), -1.0f, 1.0f);
382 buffer_[writePos_] = static_cast<int16_t>(monoIn * 32767.0f);
383
384 // P:$0FC7 - $0FD6: Advance slow 3-phase LFO (0, 1/3, 2/3 via Y:$0328 = $555555) and fast shimmer LFO (~6 Hz)
385 slowPhase_ += static_cast<float>(lfoSpeedHz_ / sampleRate_);
386 if (slowPhase_ >= 1.0f)
387 slowPhase_ -= 1.0f;
388
389 fastPhase_ += static_cast<float>((lfoSpeedHz_ * 5.8f) / sampleRate_);
390 if (fastPhase_ >= 1.0f)
391 fastPhase_ -= 1.0f;
392
393 auto triWave = [](float p) noexcept -> float
394 {
395 p -= std::floor(p);
396 return 2.0f * std::abs(2.0f * p - 1.0f) - 1.0f;
397 };
398
399 const float shimmer = 0.20f * triWave(fastPhase_); // Y:$0329 = $199999 (0.20)
400 const float mod1 = triWave(slowPhase_) + shimmer;
401 const float mod2 = triWave(slowPhase_ + (1.0f / 3.0f)) + shimmer;
402 const float mod3 = triWave(slowPhase_ + (2.0f / 3.0f)) - shimmer;
403
404 // Center delay ~ 240 samples (5.0 ms @ 48 kHz), modulated by depth
405 const float centerDelay = 240.0f * srRatio;
406 const float modSpan = (12.0f + depth_ * 140.0f) * srRatio;
407
408 const float tap1 = readTap(centerDelay + mod1 * modSpan);
409 const float tap2 = readTap(centerDelay + mod2 * modSpan);
410 const float tap3 = readTap(centerDelay + mod3 * modSpan);
411
412 writePos_ = (writePos_ + 1) & kBufferMask;
413
414 // P:$0FAA - $0FB6: Stereo Ensemble Mixing Matrix (Y:$032A = 0.4, Y:$032B = 0.8)
415 const float wetL = 0.8f * tap1 + 0.4f * tap2 + 0.2f * tap3;
416 const float wetR = 0.2f * tap1 + 0.4f * tap2 + 0.8f * tap3;
417
418 left = DSPUtils::clamp(left * (1.0f - 0.5f * mix_) + wetL * mix_, -1.0f, 1.0f);
419 right = DSPUtils::clamp(right * (1.0f - 0.5f * mix_) + wetR * mix_, -1.0f, 1.0f);
420 }
421
422 private:
423 [[nodiscard]] float readTap(float delaySamples) const noexcept
424 {
425 float d = DSPUtils::clamp(delaySamples, 2.0f, static_cast<float>(kBufferSize - 4));
426 float readPos = static_cast<float>(writePos_) - d;
427 while (readPos < 0.0f)
428 readPos += static_cast<float>(kBufferSize);
429
430 size_t idx0 = static_cast<size_t>(readPos) & kBufferMask;
431 size_t idx1 = (idx0 + 1) & kBufferMask;
432 float frac = readPos - std::floor(readPos);
433
434 float s0 = static_cast<float>(buffer_[idx0]) * (1.0f / 32768.0f);
435 float s1 = static_cast<float>(buffer_[idx1]) * (1.0f / 32768.0f);
436 return s0 + frac * (s1 - s0);
437 }
438
439 double sampleRate_ {48000.0};
440 float lfoSpeedHz_ {0.8f};
441 float depth_ {0.6f};
442 float mix_ {0.5f};
443
444 std::array<int16_t, kBufferSize> buffer_ {};
445 size_t writePos_ {0};
446 float slowPhase_ {0.0f};
447 float fastPhase_ {0.0f};
448};

450//==============================================================================
451// 5. ModFX 2: 4-Stage Stereo Phaser (DSP56362 P:$105D - $10A7)
452//==============================================================================
453/**
454 * @brief 4-Stage Stereo All-Pass Phaser extracted from `P:$105D - $10A7` (`r3 = $0260`).
455 *
456 * Microcode Reference:
457 * - Parameter block: `X:$0260..$027F`, `Y:$0330..$0335`
458 * - Runs 100% in internal 24-bit `X:` RAM (no external DRAM used).
459 * - 4 cascaded 1st-order all-pass stages per channel (`P:$106F - $107E` Left,
460 * `P:$107F - $108F` Right).
461 * - Quadratic-shaped triangle LFO (`P:$1090 - $10A6`, `mpy x0,x0,b` at `P:$1091`)
462 * with inverted stereo LFO polarity between Left and Right channels.
463 */
464class HardwarePhaser
465{
466 public:
467 void prepare(double sampleRate) noexcept
468 {
469 sampleRate_ = DSPUtils::validateSampleRate(sampleRate);
470 reset();
471 }
472
473 void reset() noexcept
474 {
475 apStateL_.fill(0.0f);
476 apStateR_.fill(0.0f);
477 fbStateL_ = 0.0f;
478 fbStateR_ = 0.0f;
479 lfoPhase_ = 0.0f;
480 }
481
482 void setParameters(float lfoSpeedHz, float depth0to1, float feedbackBipolar, float mix0to1) noexcept
483 {
484 lfoSpeedHz_ = DSPUtils::clamp(lfoSpeedHz, 0.02f, 20.0f);
485 depth_ = DSPUtils::clamp(depth0to1, 0.0f, 1.0f);
486 feedback_ = DSPUtils::clamp(feedbackBipolar, -0.92f, 0.92f);
487 mix_ = DSPUtils::clamp(mix0to1, 0.0f, 1.0f);
488 }
489
490 void processStereo(float &left, float &right) noexcept
491 {
492 // P:$1090 - $10A6: LFO phase update & quadratic triangle shaping (`mpy x0,x0,b`)
493 lfoPhase_ += static_cast<float>(lfoSpeedHz_ / sampleRate_);
494 if (lfoPhase_ >= 1.0f)
495 lfoPhase_ -= 1.0f;
496
497 float triL = 2.0f * std::abs(2.0f * lfoPhase_ - 1.0f) - 1.0f;
498 float triR = -triL; // Stereo opposite LFO polarity (`P:$1099 eor y1,a`)
499
500 // Quadratic curve mapping to all-pass coefficient g in [0.15, 0.92]
501 // (Compare X:$026D = $E80000, X:$026E = $780000)
502 auto computeAllPassCoeff = [this](float lfoBipolar) noexcept -> float
503 {
504 float norm = DSPUtils::clamp(0.5f + 0.5f * lfoBipolar * depth_, 0.0f, 1.0f);
505 float curved = norm * norm; // P:$1091 mpy x0,x0,b
506 float minFreq = 180.0f;
507 float maxFreq = 4800.0f;
508 float fc = minFreq + curved * (maxFreq - minFreq);
509 float t = std::tan(DSPUtils::PI * fc / static_cast<float>(sampleRate_));
510 return DSPUtils::clamp((1.0f - t) / (1.0f + t), -0.95f, 0.95f);
511 };
512
513 const float gL = computeAllPassCoeff(triL);
514 const float gR = computeAllPassCoeff(triR);
515
516 // P:$1063 - $1065: Input minus feedback (`mac -x1,x0,a` / `mac -x1,x0,b`)
517 float uL = DSPUtils::clamp(left - feedback_ * fbStateL_, -1.0f, 1.0f);
518 float uR = DSPUtils::clamp(right - feedback_ * fbStateR_, -1.0f, 1.0f);
519
520 // P:$106F - $107E: 4-Stage 1st-Order All-Pass Cascade (Left)
521 for (size_t i = 0; i < 4; ++i)
522 {
523 float v = DSPUtils::clamp(uL + gL * apStateL_[i], -1.0f, 1.0f);
524 uL = DSPUtils::clamp(apStateL_[i] - gL * v, -1.0f, 1.0f);
525 apStateL_[i] = v;
526 }
527
528 // P:$107F - $108F: 4-Stage 1st-Order All-Pass Cascade (Right)
529 for (size_t i = 0; i < 4; ++i)
530 {
531 float v = DSPUtils::clamp(uR + gR * apStateR_[i], -1.0f, 1.0f);
532 uR = DSPUtils::clamp(apStateR_[i] - gR * v, -1.0f, 1.0f);
533 apStateR_[i] = v;
534 }
535
536 fbStateL_ = uL;
537 fbStateR_ = uR;
538
539 // P:$1068 - $106D: Dry/Wet Mix
540 left = DSPUtils::clamp(left + mix_ * uL, -1.0f, 1.0f);
541 right = DSPUtils::clamp(right + mix_ * uR, -1.0f, 1.0f);
542 }
543
544 private:
545 double sampleRate_ {48000.0};
546 float lfoSpeedHz_ {0.5f};
547 float depth_ {0.7f};
548 float feedback_ {0.5f};
549 float mix_ {0.5f};
550
551 std::array<float, 4> apStateL_ {};
552 std::array<float, 4> apStateR_ {};
553 float fbStateL_ {0.0f};
554 float fbStateR_ {0.0f};
555 float lfoPhase_ {0.0f};
556};

558//==============================================================================
559// 6. Global Delay: Stereo / Cross / L-R Delay (DSP56362 P:$0FE9 - $105C)
560//==============================================================================
561/**
562 * @brief Stereo, Cross, and L/R Delay extracted from `P:$0FE9 - $1024` (`r3 = $0200`)
563 * and `P:$1025 - $105C` (`r3 = $0280`).
564 *
565 * Microcode Reference:
566 * - External DRAM buffers:
567 * - Left: `Y:$400000..$43FFFF` (`262,144` x 16-bit samples = `5.46 s` @ 48 kHz, mask `$03FFFF`)
568 * - Right: `Y:$440000..$47FFFF` (`262,144` x 16-bit samples = `5.46 s` @ 48 kHz)
569 * - 1-pole exponential delay-time glide (`P:$0FED - $0FF4`) for smooth tape-style pitch sweeps.
570 * - 6-bit fractional sample linear interpolation (`P:$0FFB - $0FFE`, `lsl #$12,b`, `lsr b`).
571 * - 1-pole feedback damping filter (`P:$1001 - $1006`, `Y:$0300 = $300000` [0.375],
572 * `Y:$0301 = $0E0000` [0.109375]).
573 */
574enum class HardwareDelayType
575{
576 StereoDelay = 0, // P:$0FE9 with straight L->L, R->R feedback
577 CrossDelay = 1, // P:$0FE9 with crossed L->R, R->L feedback
578 LRDelay = 2 // P:$1025 with mono-crossed feedback & split L/R output taps
579};
580
581class HardwareDelay
582{
583 public:
584 static constexpr size_t kBufferSize = 262144; // $03FFFF + 1 words per channel
585 static constexpr size_t kBufferMask = kBufferSize - 1;
586
587 void prepare(double sampleRate) noexcept
588 {
589 sampleRate_ = DSPUtils::validateSampleRate(sampleRate);
590 bufferL_.assign(kBufferSize, 0);
591 bufferR_.assign(kBufferSize, 0);
592 reset();
593 }
594
595 void reset() noexcept
596 {
597 if (!bufferL_.empty())
598 std::fill(bufferL_.begin(), bufferL_.end(), int16_t {0});
599 if (!bufferR_.empty())
600 std::fill(bufferR_.begin(), bufferR_.end(), int16_t {0});
601 writePos_ = 0;
602 smoothDelayL_ = targetDelayL_;
603 smoothDelayR_ = targetDelayR_;
604 dampStateL_ = 0.0f;
605 dampStateR_ = 0.0f;
606 }
607
608 void setParameters(
609 HardwareDelayType type, float delayTimeMsL, float delayTimeMsR, float feedback0to1, float mix0to1) noexcept
610 {
611 type_ = type;
612 const float maxMs = static_cast<float>((kBufferSize - 8) * 1000.0 / sampleRate_);
613 targetDelayL_ = DSPUtils::clamp(delayTimeMsL, 0.5f, maxMs) * 0.001f * static_cast<float>(sampleRate_);
614 targetDelayR_ = DSPUtils::clamp(delayTimeMsR, 0.5f, maxMs) * 0.001f * static_cast<float>(sampleRate_);
615 feedback_ = DSPUtils::clamp(feedback0to1, 0.0f, 0.95f);
616 mix_ = DSPUtils::clamp(mix0to1, 0.0f, 1.0f);
617 }
618
619 void processStereo(float &left, float &right) noexcept
620 {
621 if (bufferL_.size() != kBufferSize)
622 return;
623
624 // P:$0FED - $0FF4: 1-pole delay time smoothing for click-free tape-style sweeps
625 constexpr float kTimeGlide = 0.0015f;
626 smoothDelayL_ += kTimeGlide * (targetDelayL_ - smoothDelayL_);
627 smoothDelayR_ += kTimeGlide * (targetDelayR_ - smoothDelayR_);
628
629 // P:$0FFB - $1000: Read 16-bit DRAM with 6-bit fractional linear interpolation
630 const float tapL = readInterpolated6Bit(bufferL_, smoothDelayL_);
631 const float tapR = readInterpolated6Bit(bufferR_, smoothDelayR_);
632
633 // P:$1001 - $1006: 1-pole high-frequency damping in feedback loop
634 // Constants from Y:$0300..$0301: $300000 (0.375) and $0E0000 (0.109375) -> ~0.774 / 0.226 normalized
635 dampStateL_ = 0.774f * tapL + 0.226f * dampStateL_;
636 dampStateR_ = 0.774f * tapR + 0.226f * dampStateR_;
637
638 float fbL = 0.0f;
639 float fbR = 0.0f;
640 if (type_ == HardwareDelayType::StereoDelay)
641 {
642 fbL = left + feedback_ * dampStateL_;
643 fbR = right + feedback_ * dampStateR_;
644 }
645 else if (type_ == HardwareDelayType::CrossDelay)
646 {
647 fbL = left + feedback_ * dampStateR_;
648 fbR = right + feedback_ * dampStateL_;
649 }
650 else // HardwareDelayType::LRDelay (P:$1025 - $105C)
651 {
652 const float monoIn = 0.5f * (left + right);
653 fbL = monoIn + feedback_ * dampStateR_;
654 fbR = monoIn + feedback_ * dampStateL_;
655 }
656
657 bufferL_[writePos_] = quantize16(fbL);
658 bufferR_[writePos_] = quantize16(fbR);
659 writePos_ = (writePos_ + 1) & kBufferMask;
660
661 // P:$1004 - $1007: Dry + Wet Output Accumulation
662 left = DSPUtils::clamp(left + mix_ * tapL, -1.0f, 1.0f);
663 right = DSPUtils::clamp(right + mix_ * tapR, -1.0f, 1.0f);
664 }
665
666 private:
667 static int16_t quantize16(float x) noexcept
668 {
669 float clamped = DSPUtils::clamp(x, -1.0f, 0.999969f);
670 return static_cast<int16_t>(clamped * 32768.0f);
671 }
672
673 [[nodiscard]] float readInterpolated6Bit(const std::vector<int16_t> &buf, float delaySamples) const noexcept
674 {
675 // P:$0FFB - $0FFC: `lsl #$12,b` + `lsr b` extracts 6-bit fractional part (64 steps)
676 float readPos = static_cast<float>(writePos_) - delaySamples;
677 while (readPos < 0.0f)
678 readPos += static_cast<float>(kBufferSize);
679
680 size_t idx0 = static_cast<size_t>(readPos) & kBufferMask;
681 size_t idx1 = (idx0 + 1) & kBufferMask;
682 float frac = std::floor((readPos - std::floor(readPos)) * 64.0f) * (1.0f / 64.0f);
683
684 float s0 = static_cast<float>(buf[idx0]) * (1.0f / 32768.0f);
685 float s1 = static_cast<float>(buf[idx1]) * (1.0f / 32768.0f);
686 return s0 + frac * (s1 - s0);
687 }
688
689 double sampleRate_ {48000.0};
690 HardwareDelayType type_ {HardwareDelayType::StereoDelay};
691 float targetDelayL_ {12000.0f};
692 float targetDelayR_ {12000.0f};
693 float smoothDelayL_ {12000.0f};
694 float smoothDelayR_ {12000.0f};
695 float feedback_ {0.35f};
696 float mix_ {0.35f};
697 float dampStateL_ {0.0f};
698 float dampStateR_ {0.0f};
699
700 std::vector<int16_t> bufferL_;
701 std::vector<int16_t> bufferR_;
702 size_t writePos_ {0};
703};
