11/**
12 * @brief SysEx **propio de ABDSynths** para los presets del plugin (bloque nativo).
13 *
14 * Spec canónica del formato: `DOCS/ABDSynths_SysEx_Spec.md`.
15 *
16 * Por qué existe: el bloque de programa del plugin (`UNPACKED_PROGRAM_SIZE` = 384 B,
17 * layout denso del motor, hoy v2) **no**
18 * es el programa del MS2000 físico (254 B, ver `MS2000HardwareProgram.h`). Emitirlo con
19 * la cabecera de Korg (`42 3n 58`) sería disfrazar un formato de otro: un MS2000 real lo
20 * recibiría y lo interpretaría mal, y la biblioteca del Bank Manager lo abriría con el
21 * tamaño y los offsets equivocados. Así que el formato propio se identifica como lo que
22 * es: **fabricante ABDSynths, modelo propio**.
23 *
24 * Identificadores (los mismos que declara el contrato del host en ABDBankManager, que se
25 * genera en `Source/Plugin/HostModelId.gen.h`; el id no se repite aquí a propósito):
26 * - Fabricante `0x7D`: reservado por la especificación MIDI 1.0 para uso
27 * **no comercial / educativo** (escuelas, investigación, proyectos abiertos), sin
28 * registro ni cuota. No es válido para un producto publicado: si ABDSynths se
29 * comercializa habría que registrar un ID de 3 bytes (p. ej. `00 20 xx`).
30 * - Modelo `0x0A`: primer modelo de la casa (ABD MS2000). Los
31 * siguientes van detrás.
32 *
33 * Trama: `F0 7D 0A [comando] [payload 7→8] F7`
34 * - `0x40` Program Data Dump (un preset nativo de 384 B → payload de 439 B)
35 * - `0x4C` All Data Dump (N presets nativos)
36 * - `0x10` Program Dump Request, `0x0E` All Data Dump Request
37 * - `0x23` Write Completed (ACK), `0x24` Write Error (NACK): acuse de 5 B que el
38 * receptor devuelve tras **guardar** (o no) un `0x40`/`0x4C`. Llega **después**
39 * del dump, nunca dentro, así que no rompe a un emisor que no lo espere.
40 *
41 * El empaquetado 7→8 es el de siempre (`abd::hw::SysExCodec`), con el último grupo
42 * parcial sin rellenar.
43 */
44struct ABDSynthsSysEx
45{
46 static constexpr uint8_t MANUFACTURER_ID = 0x7D;
47 static constexpr uint8_t MODEL_BYTE = 0x0A;
48
49 static constexpr uint8_t CMD_PROGRAM_DUMP = 0x40;
50 static constexpr uint8_t CMD_ALL_DUMP = 0x4C;
51 static constexpr uint8_t CMD_PROGRAM_REQUEST = 0x10;
52 static constexpr uint8_t CMD_ALL_REQUEST = 0x0E;
53
54 // Acuse de escritura (los mismos códigos que usa el MS2000 real, §2 de
55 // `MS2000_SysEx_Spec.md`): `0x23` Write Completed, `0x24` Write Error.
56 static constexpr uint8_t CMD_WRITE_COMPLETED = 0x23;
57 static constexpr uint8_t CMD_WRITE_ERROR = 0x24;
58
59 /** Longitud del payload 7→8 de un bloque nativo (384 B → 439 B). */
60 static size_t packedProgramPayloadSize()
61 {
62 const size_t n = MS2000ProgramData::UNPACKED_PROGRAM_SIZE;
63 return (n / 7) * 8 + ((n % 7 == 0) ? 0 : (1 + (n % 7)));
64 }
65
66 /** Longitud de la trama de un preset nativo: `F0 7D 0A [cmd] [payload] F7`. */
67 static size_t programFrameSize() { return 4 + packedProgramPayloadSize() + 1; }
68
69 /** ¿Es un acuse de escritura de la casa (`F0 7D 0A 23/24 F7`, 5 B)? */
70 static bool isWriteAcknowledgement(const uint8_t* data, size_t size) noexcept
71 {
72 return isABDSynthsSysEx(data, size) && size == 5
73 && (data[3] == CMD_WRITE_COMPLETED || data[3] == CMD_WRITE_ERROR);
74 }
75
76 /** ¿Es una trama de ABDSynths (y de este modelo)? Acepta las peticiones de 5 B
77 * (`F0 7D 0A [cmd] F7`) además de las tramas con payload. */
78 static bool isABDSynthsSysEx(const uint8_t* data, size_t size) noexcept
79 {
80 return data != nullptr && size >= 5
81 && data[0] == 0xF0
82 && data[1] == MANUFACTURER_ID
83 && data[2] == MODEL_BYTE
84 && data[size - 1] == 0xF7;
85 }

87 /** `F0 7D 0A 40 [bloque nativo empaquetado] F7` desde un preset nativo. */
88 static std::vector<uint8_t> buildProgramDump(const MS2000ProgramData& program)
89 {
90 std::vector<uint8_t> packed;
91 if (!SysExCodec::pack8to7(program.rawData.data(), MS2000ProgramData::UNPACKED_PROGRAM_SIZE, packed))
92 return {};
93
94 std::vector<uint8_t> sysex;
95 sysex.reserve(5 + packed.size());
96 sysex.push_back(0xF0);
97 sysex.push_back(MANUFACTURER_ID);
98 sysex.push_back(MODEL_BYTE);
99 sysex.push_back(CMD_PROGRAM_DUMP);
100 sysex.insert(sysex.end(), packed.begin(), packed.end());
101 sysex.push_back(0xF7);
102 return sysex;
103 }
104
105 /** `F0 7D 0A 4C [N bloques nativos empaquetados] F7` desde una memoria completa del plugin. */
106 template <typename Container>
107 static std::vector<uint8_t> buildBankDump(const Container& programs)
108 {
109 std::vector<uint8_t> allRaw;
110 allRaw.reserve(programs.size() * MS2000ProgramData::UNPACKED_PROGRAM_SIZE);
111 for (const auto& program : programs)
112 allRaw.insert(allRaw.end(), program.rawData.begin(), program.rawData.end());
113
114 std::vector<uint8_t> packed;
115 if (allRaw.empty() || !SysExCodec::pack8to7(allRaw.data(), allRaw.size(), packed))
116 return {};
117
118 std::vector<uint8_t> sysex;
119 sysex.reserve(5 + packed.size());
120 sysex.push_back(0xF0);
121 sysex.push_back(MANUFACTURER_ID);
122 sysex.push_back(MODEL_BYTE);
123 sysex.push_back(CMD_ALL_DUMP);
124 sysex.insert(sysex.end(), packed.begin(), packed.end());
125 sysex.push_back(0xF7);
126 return sysex;
127 }
128
129 /** `F0 7D 0A [cmd] F7` */
130 static std::vector<uint8_t> buildRequest(uint8_t command) { return { 0xF0, MANUFACTURER_ID, MODEL_BYTE, command, 0xF7 }; }
131
132 /** `F0 7D 0A 23 F7` — el bloque llegó y se guardó. */
133 static std::vector<uint8_t> buildWriteCompleted() { return buildRequest(CMD_WRITE_COMPLETED); }
134
135 /** `F0 7D 0A 24 F7` — el bloque llegó pero no se pudo guardar. */
136 static std::vector<uint8_t> buildWriteError() { return buildRequest(CMD_WRITE_ERROR); }
137};
