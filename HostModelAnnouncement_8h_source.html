41/**
42 * Acciones `webuiToCpp` que obligan a anunciar el modelId del host:
43 * - `requestState`: lo envía el ABD Bank Manager embebido cuando su bridge ya
44 * está suscrito a `hostModel` (ABDBankManager/WebUI/src/app.js hace
45 * `initHostContext(bridge)` antes de `bridge.send('requestState')`), por lo
46 * que es el momento fiable de entrega.
47 * - `requestFullState`: handshake inicial de la WebUI anfitriona. Es
48 * idempotente y cubre el orden inverso de carga.
49 */
50inline bool actionAnnouncesHostModel(const juce::String& action)
51{
52 return action == "requestState" || action == "requestFullState";
53}

70/**
71 * Mensaje `cppToWebui` de la acción `hostModel`:
72 * `{ type, data: { modelId, displayName, manufacturer, protocol, buildStamp, buildRevision } }`.
73 *
74 * El `modelId` es lo que consume el filtrado; los cuatro campos de sello viajan
75 * para que el Bank Manager pueda registrar/diagnosticar contra qué build habla
76 * (y un build antiguo del Bank Manager los ignora sin romperse).
77 */
78inline juce::var hostModelMessage()
79{
80 juce::DynamicObject::Ptr data = new juce::DynamicObject();
81 data->setProperty("modelId", hostModelId());
82 data->setProperty("displayName", juce::String(kHostModelDisplayName));
83 data->setProperty("manufacturer", juce::String(kHostModelManufacturer));
84 data->setProperty("protocol", kHostBridgeProtocol);
85 data->setProperty("buildStamp", hostBuildStamp());
86 data->setProperty("buildRevision", hostBuildRevision());
87
88 juce::DynamicObject::Ptr message = new juce::DynamicObject();
89 message->setProperty("type", "hostModel");
90 message->setProperty("data", juce::var(data.get()));
91 return juce::var(message.get());
92}

94/**
