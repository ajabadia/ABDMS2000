17/**
18 * @brief Korg MS2000 Program Data — 128-byte Sysex layout with registry-driven encoding.
19 *
20 * Byte Layout (128 bytes):
21 * [0x00..0x0B] Program Name (12 bytes, ASCII)
22 * [0x0C] Voice Mode byte (packed: voiceMode | unisonDetune | portamentoTime | portamentoOn)
23 * [0x0D] Reserved
24 * [0x0E..0x4B] Timbre parameters — byte positions driven by ParameterRegistry sysexOffset
25 * [0x4C..0x7F] Reserved (future: all-data-dump extensions)
26 *
27 * Conversion between 8-bit rawData and APVTS is fully automatic:
28 * it iterates ParameterRegistry::getAllParameters() and reads/writes bytes
29 * at the sysexOffset position for each parameter that has a valid offset.
30 *
31 * Special-case signed parameters (bias = 64) are hardcoded in the registry
32 * via their min/max ranges and handled by a small lookup set.
33 */
34struct MS2000ProgramData {
35 static constexpr size_t UNPACKED_PROGRAM_SIZE = 128;
36 static constexpr size_t NAME_LENGTH = 12;
37 static constexpr uint8_t NAME_END = 12;
38 static constexpr uint8_t VOICE_BYTE = 0x0C;
39 static constexpr uint8_t TIMBRE_START = 0x0E;
40
41 std::array<uint8_t, UNPACKED_PROGRAM_SIZE> rawData{};
42
43 MS2000ProgramData() { reset(); }
44
45 /** Resets to the canonical "Init Synth" state. */
46 void reset()
47 {
48 rawData.fill(0);
49 setName("Init Synth");
50
51 // Voice defaults
52 rawData[VOICE_BYTE] = 1; // Poly (mode 1)
53 rawData[VOICE_BYTE + 2] = 0; // Portamento time 0
54
55 // Populate default values from ParameterRegistry
56 for (const auto& meta : ParameterRegistry::getAllParameters())
57 {
58 if (meta.sysexOffset < 0 || isVoiceByteParam(meta.id)) continue;
59 const auto off = TIMBRE_START + static_cast<size_t>(meta.sysexOffset);
60 if (off >= UNPACKED_PROGRAM_SIZE) continue;
61
62 if (meta.type == ParamType::Boolean)
63 rawData[off] = (meta.defaultValue > 0.5f) ? 64 : 0;
64 else if (meta.min < 0.0f)
65 rawData[off] = static_cast<uint8_t>(static_cast<int>(meta.defaultValue) + 64);
66 else
67 rawData[off] = static_cast<uint8_t>(meta.defaultValue);
68 }
69 }

73 void setName(const std::string& newName)
74 {
75 std::string padded = newName.substr(0, NAME_LENGTH);
76 while (padded.length() < NAME_LENGTH) padded += ' ';
77 for (size_t i = 0; i < NAME_LENGTH; ++i)
78 rawData[i] = static_cast<uint8_t>(padded[i]);
79 }

81 std::string getName() const
82 {
83 std::string res;
84 res.reserve(NAME_LENGTH);
85 for (size_t i = 0; i < NAME_LENGTH; ++i)
86 {
87 char c = static_cast<char>(rawData[i]);
88 res += (c >= 32 && c <= 126) ? c : ' ';
89 }
90 return res;
91 }

101 /** Convenient parameter setter by ID */
102 void setParam(const char* id, float value)
103 {
104 if (std::strcmp(id, ParamIDs::voiceMode) == 0)
105 {
106 rawData[VOICE_BYTE] = (rawData[VOICE_BYTE] & ~0x03) | (static_cast<uint8_t>(value) & 0x03);
107 return;
108 }
109 if (std::strcmp(id, ParamIDs::unisonDetune) == 0)
110 {
111 rawData[VOICE_BYTE] = (rawData[VOICE_BYTE] & ~(0x0F << 2)) | ((static_cast<uint8_t>(value) & 0x0F) << 2);
112 return;
113 }
114 if (std::strcmp(id, ParamIDs::portamentoOn) == 0)
115 {
116 rawData[VOICE_BYTE] = (rawData[VOICE_BYTE] & ~(0x01 << 6)) | ((static_cast<uint8_t>(value > 0.5f ? 1 : 0) & 0x01) << 6);
117 return;
118 }
119 if (std::strcmp(id, ParamIDs::portamentoTime) == 0)
120 {
121 rawData[VOICE_BYTE + 2] = static_cast<uint8_t>(value);
122 return;
123 }
124
125 const auto* meta = ParameterRegistry::getParameter(id);
126 if (!meta || meta->sysexOffset < 0) return;
127 const auto off = TIMBRE_START + static_cast<size_t>(meta->sysexOffset);
128 if (off >= UNPACKED_PROGRAM_SIZE) return;
129
130 if (meta->type == ParamType::Boolean)
131 rawData[off] = (value > 0.5f) ? 64 : 0;
132 else if (meta->min < 0.0f)
133 rawData[off] = static_cast<uint8_t>(static_cast<int>(value) + 64);
134 else
135 rawData[off] = static_cast<uint8_t>(value);
136 }

149 void applyToAPVTS(juce::AudioProcessorValueTreeState& apvts) const
150 {
151 // 1. Unpack the voice byte
152 const uint8_t voiceByte = rawData[VOICE_BYTE];
153 auto setChoice = [&](const char* id, float rawVal) {
154 if (auto* p = apvts.getParameter(id))
155 p->setValueNotifyingHost(p->convertTo0to1(rawVal));
156 };
157 setChoice(ParamIDs::voiceMode, static_cast<float>(voiceByte & 0x03));
158 setChoice(ParamIDs::unisonDetune, static_cast<float>((voiceByte >> 2) & 0x0F));
159 setChoice(ParamIDs::portamentoOn, static_cast<float>((voiceByte >> 6) & 0x01));
160
161 if (auto* p = apvts.getParameter(ParamIDs::portamentoTime))
162 p->setValueNotifyingHost(p->convertTo0to1(static_cast<float>(rawData[VOICE_BYTE + 2])));
163
164 // 2. Iterate registry for all other timbre parameters
165 for (const auto& meta : ParameterRegistry::getAllParameters())
166 {
167 if (meta.sysexOffset < 0) continue;
168 if (isVoiceByteParam(meta.id)) continue;
169 const auto off = TIMBRE_START + static_cast<size_t>(meta.sysexOffset);
170 if (off >= UNPACKED_PROGRAM_SIZE) continue;
171
172 float value;
173 if (meta.type == ParamType::Boolean)
174 {
175 value = (rawData[off] >= 64) ? 1.0f : 0.0f;
176 }
177 else if (meta.type == ParamType::Choice)
178 {
179 value = static_cast<float>(rawData[off]);
180 }
181 else if (meta.min < 0.0f)
182 {
183 // Signed parameters stored as unsigned with +64 bias
184 value = static_cast<float>(static_cast<int>(rawData[off]) - 64);
185 }
186 else
187 {
188 value = static_cast<float>(rawData[off]);
189 }
190
191 if (auto* p = apvts.getParameter(meta.id))
192 p->setValueNotifyingHost(p->convertTo0to1(value));
193 }

196 void extractFromAPVTS(const juce::AudioProcessorValueTreeState& apvts,
197 const std::string& progName)
198 {
199 setName(progName);
200
201 // 1. Pack voice parameters into voiceByte
202 auto getChoice = [&](const char* id) -> uint8_t {
203 if (auto* p = apvts.getRawParameterValue(id))
204 return static_cast<uint8_t>(p->load());
205 return 0;
206 };
207 rawData[VOICE_BYTE] = (getChoice(ParamIDs::voiceMode) & 0x03)
208 | ((getChoice(ParamIDs::unisonDetune) & 0x0F) << 2)
209 | ((getChoice(ParamIDs::portamentoOn) & 0x01) << 6);
210
211 if (auto* p = apvts.getRawParameterValue(ParamIDs::portamentoTime))
212 rawData[VOICE_BYTE + 2] = static_cast<uint8_t>(p->load());
213
214 // 2. Iterate registry for all other timbre parameters
215 for (const auto& meta : ParameterRegistry::getAllParameters())
216 {
217 if (meta.sysexOffset < 0) continue;
218 if (isVoiceByteParam(meta.id)) continue;
219 const auto off = TIMBRE_START + static_cast<size_t>(meta.sysexOffset);
220 if (off >= UNPACKED_PROGRAM_SIZE) continue;
221
222 float native = 0.0f;
223 if (auto* p = apvts.getRawParameterValue(meta.id))
224 native = p->load();
225
226 if (meta.type == ParamType::Boolean)
227 {
228 rawData[off] = (native > 0.5f) ? 64 : 0;
229 }
230 else if (meta.type == ParamType::Choice)
231 {
232 rawData[off] = static_cast<uint8_t>(native);
233 }
234 else if (meta.min < 0.0f)
235 {
236 // Signed → unsigned with +64 bias
237 rawData[off] = static_cast<uint8_t>(static_cast<int>(native) + 64);
238 }
239 else
240 {
241 rawData[off] = static_cast<uint8_t>(native);
242 }
243 }

249 /**
250 * Unpacks a 7-bit encoded SysEx payload into this 128-byte rawData buffer.
251 * @return true on success.
252 */
253 bool unpackFromSysexPayload(const uint8_t* payload, size_t payloadLen)
254 {
255 std::vector<uint8_t> unpacked;
256 if (!SysExCodec::unpack7to8(payload, payloadLen, unpacked) || unpacked.empty())
257 return false;
258 const size_t n = std::min(unpacked.size(), UNPACKED_PROGRAM_SIZE);
259 std::copy_n(unpacked.begin(), n, rawData.begin());
260 return true;
261 }
262
263 /**
264 * Packs this 128-byte rawData buffer into a 7-bit SysEx payload.
265 * @return true on success.

267 bool packToSysexPayload(std::vector<uint8_t>& outPayload) const
268 {
269 return SysExCodec::pack8to7(rawData.data(), UNPACKED_PROGRAM_SIZE, outPayload);

274 /**
275 * Validates a raw 128-byte buffer as a Korg MS2000 Program Data block.
276 * @return empty string on success, error description on failure.
277 */
278 std::string validate() const
279 {
280 // Check name bytes are printable
281 for (size_t i = 0; i < NAME_LENGTH; ++i)
282 {
283 if (rawData[i] > 127) return "Name byte " + std::to_string(i) + " out of range";
284 }
285 // Voice mode (byte 0) must be 0..2
286 if (rawData[VOICE_BYTE] > 2)
287 return "Voice mode byte out of range (expected 0..2, got " +
288 std::to_string(rawData[VOICE_BYTE]) + ")";
289 return {};
290 }

308 /**
309 * Formats the 128-byte raw data as a 2-column hex + ASCII dump string.
310 */
311 std::string toHexDump(size_t bytesPerLine = 16) const
312 {
313 std::ostringstream oss;
314 oss << "MS2000 Program: " << getName() << "\n";
315 oss << "Voice Mode: " << getVoiceModeName() << "\n\n";
316
317 size_t offset = 0;
318 while (offset < UNPACKED_PROGRAM_SIZE)
319 {
320 oss << std::hex << std::setw(3) << std::setfill('0') << offset << ": ";
321 size_t lineBytes = std::min(bytesPerLine, UNPACKED_PROGRAM_SIZE - offset);
322 for (size_t i = 0; i < bytesPerLine; ++i)
323 {
324 if (i < lineBytes)
325 oss << std::hex << std::setw(2) << std::setfill('0')
326 << static_cast<int>(rawData[offset + i]) << " ";
327 else oss << " ";
328 if (i == 7) oss << " ";
329 }
330 oss << " |";
331 for (size_t i = 0; i < lineBytes; ++i)
332 {
333 char c = static_cast<char>(rawData[offset + i]);
334 oss << ((c >= 32 && c <= 126) ? c : '.');
335 }
336 oss << "|\n";
337 offset += lineBytes;
338 }
339 return oss.str();
340 }
