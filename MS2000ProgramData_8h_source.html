20/**
21 * @brief Programa **propio** del plugin (el sistema `native` que declara el contrato
22 * del host, ver `Source/Plugin/HostModelId.gen.h`):
23 * bloque de 384 B con codificación dirigida por el `ParameterRegistry`.
24 *
25 * Byte Layout (384 bytes — v2):
26 * [0x00..0x0B] Program Name (12 bytes, ASCII)
27 * [0x0C] Voice Mode byte (packed: voiceMode | unisonDetune | portamentoTime | portamentoOn)
28 * [0x0D] Reserved
29 * [0x0E..0x10A] Parámetros — byte dirigido por el `sysexOffset` del registro (1 byte por
30 * parámetro marcado `sysex != false`, hoy 253). Cubre Timbre 1 **y Timbre 2**,
31 * velocidades, escala/split y los pasos del mod sequence.
32 * [0x10B..0x17F] Reservado
33 *
34 * La v1 (128 B) solo cabía Timbre 1 + voz: los 30 parámetros marcados `sysex: false`
35 * (vocoder entero, `modSeqOn/Type`, `unisonSpread`, `masterVolume`…) ya se perdían, y
36 * Timbre 2 ni existía. La v2 es el mismo
37 * layout al principio, así que **un bloque v1 se sigue leyendo** (`unpackFromSysexPayload`
38 * rellena con ceros el resto). El bloque es opaco para el ABD Bank Manager: el tamaño
39 * no viaja en el contrato, solo en la carga útil base64.
40 *
41 * Conversion between 8-bit rawData and APVTS is fully automatic:
42 * it iterates ParameterRegistry::getAllParameters() and reads/writes bytes
43 * at the sysexOffset position for each parameter that has a valid offset.
44 *
45 * Special-case signed parameters (bias = 64) are hardcoded in the registry
46 * via their min/max ranges and handled by a small lookup set.
47 */
48struct MS2000ProgramData
49{

198 res.reserve(NAME_LENGTH);

225 /** Convenient parameter setter by ID */
226 void setParam(const char *id, float value)
227 {
228 if (std::strcmp(id, ParamIDs::voiceMode) == 0)
229 {
230 rawData[VOICE_BYTE] = (rawData[VOICE_BYTE] & ~0x03) | (static_cast<uint8_t>(value) & 0x03);
231 return;
232 }
233 if (std::strcmp(id, ParamIDs::unisonDetune) == 0)
234 {
235 const uint8_t u = static_cast<uint8_t>(std::clamp(static_cast<int>(value), 0, 99));
236 rawData[VOICE_BYTE] = (rawData[VOICE_BYTE] & ~(0x0F << 2)) | ((u & 0x0F) << 2);
237 rawData[TIMBRE_START + 1] = u;
238 return;
239 }
240 if (std::strcmp(id, ParamIDs::portamentoOn) == 0)
241 {
242 rawData[VOICE_BYTE] =
243 (rawData[VOICE_BYTE] & ~(0x01 << 6)) | ((static_cast<uint8_t>(value > 0.5f ? 1 : 0) & 0x01) << 6);
244 return;
245 }
246 if (std::strcmp(id, ParamIDs::portamentoTime) == 0)
247 {
248 rawData[VOICE_BYTE + 2] = static_cast<uint8_t>(value);
249 return;
250 }
251
252 const auto *meta = ParameterRegistry::getParameter(id);
253 if (!meta)
254 return;
255 const int offInt = getStorageOffsetById(id);
256 if (offInt < 0 || static_cast<size_t>(offInt) >= UNPACKED_PROGRAM_SIZE)
257 return;
258 const auto off = static_cast<size_t>(offInt);
259
260 if (meta->type == ParamType::Boolean)
261 rawData[off] = (value > 0.5f) ? 64 : 0;
262 else if (meta->type == ParamType::Continuous && meta->max <= 1.0f)
263 rawData[off] = static_cast<uint8_t>(std::clamp(static_cast<int>(value * 100.0f + 0.5f), 0, 100));
264 else if (meta->min < -64.0f)
265 rawData[off] = static_cast<uint8_t>(static_cast<int8_t>(std::clamp(static_cast<int>(value), -100, 100)));
266 else if (meta->min < 0.0f)
267 rawData[off] = static_cast<uint8_t>(static_cast<int>(value) + 64);
268 else
269 rawData[off] = static_cast<uint8_t>(value);
270 }

282 template <typename Fn> void forEachParameterValue(Fn &&callback) const
283 {
284 const uint8_t voiceByte = rawData[VOICE_BYTE];
285 const uint8_t fullDetune = rawData[TIMBRE_START + 1];
286 callback(ParamIDs::voiceMode, static_cast<float>(voiceByte & 0x03));
287 callback(ParamIDs::unisonDetune, static_cast<float>(fullDetune != 0 ? fullDetune : ((voiceByte >> 2) & 0x0F)));
288 callback(ParamIDs::portamentoOn, static_cast<float>((voiceByte >> 6) & 0x01));
289 callback(ParamIDs::portamentoTime, static_cast<float>(rawData[VOICE_BYTE + 2]));
290
291 int nonSysexIdx = 0;
292 for (const auto &meta : ParameterRegistry::getAllParameters())
293 {
294 if (isVoiceByteParam(meta.id))
295 continue;
296 const int offInt = (meta.sysexOffset >= 0) ? (static_cast<int>(TIMBRE_START) + meta.sysexOffset)
297 : (static_cast<int>(NON_SYSEX_START) + (nonSysexIdx++));
298 if (offInt < 0 || static_cast<size_t>(offInt) >= UNPACKED_PROGRAM_SIZE)
299 continue;
300 const auto off = static_cast<size_t>(offInt);
301
302 float value;
303 if (meta.type == ParamType::Boolean)
304 {
305 value = (rawData[off] >= 64) ? 1.0f : 0.0f;
306 }
307 else if (meta.type == ParamType::Continuous && meta.max <= 1.0f)
308 {
309 if (std::strcmp(meta.id, ParamIDs::masterVolume) == 0 && rawData[off] == 0)
310 value = meta.defaultValue;
311 else
312 value = static_cast<float>(std::min<uint8_t>(rawData[off], 100)) / 100.0f;
313 }
314 else if (meta.type == ParamType::Choice)
315 {
316 value = static_cast<float>(rawData[off]);
317 }
318 else if (meta.min < -64.0f)
319 {
320 value = static_cast<float>(std::clamp<int>(static_cast<int8_t>(rawData[off]), -100, 100));
321 }
322 else if (meta.min < 0.0f)
323 {
324 value = static_cast<float>(static_cast<int>(rawData[off]) - 64);
325 }
326 else
327 {
328 value = static_cast<float>(rawData[off]);
329 }
330 callback(meta.id, value);
331 }

369 void extractFromAPVTS(const juce::AudioProcessorValueTreeState &apvts, const std::string &progName)
370 {
371 setName(progName);
372 if (apvts.state.hasProperty("presetTempoBPM"))
373 setTempoBPM(static_cast<int>(apvts.state.getProperty("presetTempoBPM", 120)));
374 // UX-4: lectura preferente del parametro real, con respaldo en la propiedad suelta.
375 auto paramOrProperty = [&apvts](const char *id, int fallback) -> int
376 {
377 if (auto *p = apvts.getRawParameterValue(id))
378 return static_cast<int>(std::lround(p->load()));
379 if (apvts.state.hasProperty(id))
380 return static_cast<int>(apvts.state.getProperty(id, fallback));
381 return fallback;
382 };
383 setArpTarget(static_cast<uint8_t>(paramOrProperty("arpTarget", 0)));
384 setArpKeySync(paramOrProperty("arpKeySync", 0) != 0);
385 setDelayTempoSync(paramOrProperty("delayTempoSync", 0) != 0);
386 setDelaySyncNote(static_cast<uint8_t>(paramOrProperty("delaySyncNote", 0)));
387
388 // 1. Pack voice parameters into voiceByte
389 auto getChoice = [&](const char *id) -> uint8_t
390 {
391 if (auto *p = apvts.getRawParameterValue(id))
392 return static_cast<uint8_t>(p->load());
393 return 0;
394 };
395 const uint8_t fullDetune = static_cast<uint8_t>(std::clamp<int>(getChoice(ParamIDs::unisonDetune), 0, 99));
396 rawData[VOICE_BYTE] = (getChoice(ParamIDs::voiceMode) & 0x03) | ((fullDetune & 0x0F) << 2)
397 | ((getChoice(ParamIDs::portamentoOn) & 0x01) << 6);
398 rawData[TIMBRE_START + 1] = fullDetune;
399
400 if (auto *p = apvts.getRawParameterValue(ParamIDs::portamentoTime))
401 rawData[VOICE_BYTE + 2] = static_cast<uint8_t>(p->load());

404 int nonSysexIdx = 0;
405 for (const auto &meta : ParameterRegistry::getAllParameters())
406 {

408 continue;

410 : (static_cast<int>(NON_SYSEX_START) + (nonSysexIdx++));
411 if (offInt < 0 || static_cast<size_t>(offInt) >= UNPACKED_PROGRAM_SIZE)
412 continue;
413 const auto off = static_cast<size_t>(offInt);
414
415 float native = 0.0f;
416 if (auto *p = apvts.getRawParameterValue(meta.id))
417 native = p->load();
418
419 if (meta.type == ParamType::Boolean)
420 {
421 rawData[off] = (native > 0.5f) ? 64 : 0;
422 }
423 else if (meta.type == ParamType::Continuous && meta.max <= 1.0f)
424 {
425 rawData[off] = static_cast<uint8_t>(std::clamp(static_cast<int>(native * 100.0f + 0.5f), 0, 100));
426 }
427 else if (meta.type == ParamType::Choice)
428 {
429 rawData[off] = static_cast<uint8_t>(native);
430 }
431 else if (meta.min < -64.0f)
432 {
433 // Complemento a 2: rango -100..+100 (ej. arpSwing)
434 rawData[off] = static_cast<uint8_t>(static_cast<int8_t>(std::clamp(static_cast<int>(native), -100, 100)));
435 }
436 else if (meta.min < 0.0f)
437 {
438 // Signed → unsigned with +64 bias
439 rawData[off] = static_cast<uint8_t>(static_cast<int>(native) + 64);
440 }
441 else
442 {
443 rawData[off] = static_cast<uint8_t>(native);
444 }

451 /**
452 * Unpacks a 7-bit encoded SysEx payload into this 128-byte rawData buffer.
453 * @return true on success.
454 */
455 bool unpackFromSysexPayload(const uint8_t *payload, size_t payloadLen)
456 {
457 std::vector<uint8_t> unpacked;
458 if (!SysExCodec::unpack7to8(payload, payloadLen, unpacked) || unpacked.empty())
459 return false;
460 const size_t n = std::min(unpacked.size(), UNPACKED_PROGRAM_SIZE);
461 std::copy_n(unpacked.begin(), n, rawData.begin());
462 return true;
463 }
464
465 /**
466 * Packs this 128-byte rawData buffer into a 7-bit SysEx payload.
467 * @return true on success.
468 */
469 bool packToSysexPayload(std::vector<uint8_t> &outPayload) const
470 {
471 return SysExCodec::pack8to7(rawData.data(), UNPACKED_PROGRAM_SIZE, outPayload);
472 }

476 /**
477 * Validates a raw 128-byte buffer as a Korg MS2000 Program Data block.
478 * @return empty string on success, error description on failure.
479 */
480 std::string validate() const
481 {
482 // Check name bytes are printable
483 for (size_t i = 0; i < NAME_LENGTH; ++i)
484 {
485 if (rawData[i] > 127)
486 return "Name byte " + std::to_string(i) + " out of range";
487 }
488 // Voice mode (byte 0) must be 0..2
489 if (rawData[VOICE_BYTE] > 2)
490 return "Voice mode byte out of range (expected 0..2, got " + std::to_string(rawData[VOICE_BYTE]) + ")";
491 return {};
492 }

514 /**
515 * Formats the 128-byte raw data as a 2-column hex + ASCII dump string.
516 */
517 std::string toHexDump(size_t bytesPerLine = 16) const
518 {
519 std::ostringstream oss;
520 oss << "MS2000 Program: " << getName() << "\n";
521 oss << "Voice Mode: " << getVoiceModeName() << "\n\n";
522
523 size_t offset = 0;
524 while (offset < UNPACKED_PROGRAM_SIZE)
525 {
526 oss << std::hex << std::setw(3) << std::setfill('0') << offset << ": ";
527 size_t lineBytes = std::min(bytesPerLine, UNPACKED_PROGRAM_SIZE - offset);
528 for (size_t i = 0; i < bytesPerLine; ++i)
529 {
530 if (i < lineBytes)
531 oss << std::hex << std::setw(2) << std::setfill('0') << static_cast<int>(rawData[offset + i])
532 << " ";
533 else
534 oss << " ";
535 if (i == 7)
536 oss << " ";
537 }
538 oss << " |";
539 for (size_t i = 0; i < lineBytes; ++i)
540 {
541 char c = static_cast<char>(rawData[offset + i]);
542 oss << ((c >= 32 && c <= 126) ? c : '.');
543 }
544 oss << "|\n";
545 offset += lineBytes;
546 }
547 return oss.str();
548 }
