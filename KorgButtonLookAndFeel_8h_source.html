28 auto cornerSize = 2.5f;
29
30 juce::Colour baseColor = backgroundColour;
31 if (shouldDrawButtonAsHighlighted) baseColor = baseColor.brighter(0.08f);
32 if (shouldDrawButtonAsDown) baseColor = baseColor.darker(0.15f);
33
34 // 1. Sombra proyectada (Efecto relieve 3D)
35 g.setColour(juce::Colours::black.withAlpha(0.3f));
36 g.fillRoundedRectangle(bounds.translated(0.0f, 1.5f).reduced(0.5f), cornerSize);
37
38 // 2. Cuerpo del botón (Gradiente vertical)
39 juce::ColourGradient bodyGrad(baseColor.brighter(0.12f), bounds.getX(), bounds.getY(),
40 baseColor.darker(0.22f), bounds.getX(), bounds.getBottom(), false);
41 g.setBrush(bodyGrad);
42 g.fillRoundedRectangle(bounds.reduced(1.0f), cornerSize);
43
44 // 3. Bisel interior
45 g.setColour(baseColor.brighter(0.25f).withAlpha(0.4f));
46 g.drawRoundedRectangle(bounds.reduced(1.5f), cornerSize, 1.0f);
47
48 // 4. Borde exterior
49 g.setColour(juce::Colours::black.withAlpha(0.6f));
50 g.drawRoundedRectangle(bounds.reduced(1.0f), cornerSize, 1.0f);
51
52 // 5. Diodo LED integrado
53 if (button.getToggleState() || shouldDrawButtonAsDown)
54 {
55 float ledSize = 4.5f;
56 float ledX = (bounds.getWidth() - ledSize) * 0.5f;
57 float ledY = 5.0f;
58
59 juce::Colour ledRed = juce::Colour::fromRGBA(255, 40, 20, 255);
60 g.setColour(ledRed);
61 g.fillEllipse(ledX, ledY, ledSize, ledSize);
62
63 // Glow analógico
64 g.setColour(ledRed.withAlpha(0.4f));
65 g.fillEllipse(ledX - 2.0f, ledY - 2.0f, ledSize + 4.0f, ledSize + 4.0f);
66
67 g.setColour(ledRed.withAlpha(0.15f));
68 g.fillEllipse(ledX - 4.0f, ledY - 4.0f, ledSize + 8.0f, ledSize + 8.0f);
69 }
70 else
