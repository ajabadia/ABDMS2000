54// Asymmetric distortion for MS2000 Amp Distortion
55// Pre-gains the signal heavily, applies asymmetric tanh waveshaping,
56// then applies makeup gain. The MS2000 distortion is deliberately gritty.
57inline float ampDistortion(float x, float drive) noexcept
58{
59 float preGain = 1.0f + drive * 8.0f; // Up to 7.8x pre-gain
60 float in = x * preGain;
61 float shaped;
62 // Asymmetric clipping curve (even harmonics from asymmetry)
63 if (in > 0.0f)
64 shaped = std::tanh(in * 1.5f);
65 else
66 shaped = std::tanh(in * 2.0f) * 0.75f;
67 // Makeup gain to restore perceived loudness
68 return shaped * (1.0f / (0.5f + drive * 0.5f));
69}
