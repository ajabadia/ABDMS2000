24 /**
25 * Builds a 1-program SysEx message (.syx) compatible with physical Korg MS2000 hardware.
26 * @param apvts Reference to the AudioProcessorValueTreeState containing the current patch parameters.
27 * @param programName 12-character ASCII program name to embed in the preset.
28 * @param outSysExBuffer MemoryBlock where the resulting packed SysEx message will be written.
29 * @param midiChannel Global MIDI channel (1 to 16, default 1).
30 */
31 static void exportSingleProgram(const juce::AudioProcessorValueTreeState& apvts,
32 const juce::String& programName,
33 juce::MemoryBlock& outSysExBuffer,
34 int midiChannel = 1)
35 {
36 // 1. Extract data into the 128-byte raw program buffer
37 MS2000ProgramData progData;
38 progData.extractFromAPVTS(apvts, programName.toStdString());
39
40 // 2. Empaquetar de 8-bits a 7-bits (ventanas de 7 bytes -> 1 MSB collector + 7 data bytes)
41 std::vector<uint8_t> packed7Bit;
42 SysExCodec::pack8to7(progData.rawData.data(), MS2000ProgramData::UNPACKED_PROGRAM_SIZE, packed7Bit);
43
44 // 3. Construir la cabecera oficial de Korg y volcar a MemoryBlock
45 outSysExBuffer.clear();
46 int ch = std::max(1, std::min(16, midiChannel));
47
48 outSysExBuffer.appendByte(0xF0); // Exclusive Status
49 outSysExBuffer.appendByte(0x42); // Korg ID
50 outSysExBuffer.appendByte(static_cast<uint8_t>(0x30 | (ch - 1))); // Channel format 30..3F
51 outSysExBuffer.appendByte(0x58); // Model ID: MS2000
52 outSysExBuffer.appendByte(0x40); // 1-Program Data Dump (0x40 / 0x4C)
53
54 outSysExBuffer.append(packed7Bit.data(), packed7Bit.size());
55 outSysExBuffer.appendByte(0xF7); // End of Exclusive
56 }

58 /**
59 * Exports a full bank of 128 programs to an All-Data Dump (.syx) message.
60 */
61 static void exportBank(const std::array<MS2000ProgramData, 128>& bank,
62 juce::MemoryBlock& outSysExBuffer,
63 int midiChannel = 1)
64 {
65 std::vector<uint8_t> allRaw;
66 allRaw.reserve(128 * MS2000ProgramData::UNPACKED_PROGRAM_SIZE);
67
68 for (const auto& prog : bank)
69 {
70 allRaw.insert(allRaw.end(), prog.rawData.begin(), prog.rawData.end());
71 }
72
73 std::vector<uint8_t> packed7Bit;
74 SysExCodec::pack8to7(allRaw.data(), allRaw.size(), packed7Bit);
75
76 outSysExBuffer.clear();
77 int ch = std::max(1, std::min(16, midiChannel));
78
79 outSysExBuffer.appendByte(0xF0);
80 outSysExBuffer.appendByte(0x42);
81 outSysExBuffer.appendByte(static_cast<uint8_t>(0x30 | (ch - 1)));
82 outSysExBuffer.appendByte(0x58);
83 outSysExBuffer.appendByte(0x4C); // All Data Dump
84
85 outSysExBuffer.append(packed7Bit.data(), packed7Bit.size());
86 outSysExBuffer.appendByte(0xF7);
87 }
