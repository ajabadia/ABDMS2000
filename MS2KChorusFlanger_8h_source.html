14 * @brief MS2KChorusFlanger - Efecto híbrido continuo Chorus/Flanger del Korg MS2000 (DSP56362).
15 *
16 * Características:
17 * - Transición continua gobernada por Feedback: 0 es un Chorus cálido (7.5ms base, +/-3.5ms excursión),
18 * y a 127 se convierte en un Flanger resonante tipo "jet" (1.8ms base, +/-1.5ms excursión, feedback 0.94).
19 * - LFO en contrafase para apertura estéreo amplia (L = +lfo, R = -lfo).
20 * - Buffer estático circular en array de 4096 muestras (100% Real-Time Safe, cero heap).
21 * - Interpolación cúbica Hermite de 4 puntos (sin amortiguación de frecuencias altas ni escalonamiento).
22 * - Determinismo de fase con abd::dsp::sin.
23 */

66 inline void processSample(float &left, float &right) noexcept
67 {
68 // 1. Avance del LFO determinista
69 const float lfo = abd::dsp::sin(lfoPhase_);
70 lfoPhase_ += lfoInc_;
71 if (lfoPhase_ >= Constants::kTwoPi)
72 lfoPhase_ -= Constants::kTwoPi;
73
74 // 2. Retardo base y excursión gobernados por el lazo de feedback híbrido
75 const float fbNorm = rawFeedback_ / Constants::kMidiMaxVal;
76 const float baseDelaySec = DSPUtils::kChorusBaseDelaySec - (DSPUtils::kFlangerDelayRangeSec * fbNorm);
77 const float excursionSec =
78 (DSPUtils::kChorusExcursionSec - (DSPUtils::kFlangerExcursionRangeSec * fbNorm)) * depth_;
79
80 const float baseDelaySamples = baseDelaySec * static_cast<float>(sampleRate_);
81 const float modExcursionSamples = excursionSec * static_cast<float>(sampleRate_);
82
83 const float delayL = baseDelaySamples + (lfo * modExcursionSamples);
84 const float delayR = baseDelaySamples - (lfo * modExcursionSamples); // Desfase 180° estéreo
85
86 // 3. Lectura con interpolación cúbica Hermite
87 const float wetL = readHermite(bufferL_, delayL);
88 const float wetR = readHermite(bufferR_, delayR);
89
90 // 4. Realimentación en la línea de retardo (hasta 0.94 en Flanger extremo)
91 const float fbGain = fbNorm * Constants::kFlangerMaxFeedback;
92 bufferL_[writeIndex_] = left + (wetL * fbGain);
93 bufferR_[writeIndex_] = right + (wetR * fbGain);
94 writeIndex_ = (writeIndex_ + 1) & kBufferMask;
95
96 // 5. Mezcla Wet/Dry dinámica
97 const float wetGain = depth_ * 0.75f;
98 const float dryGain = std::max(0.35f, 1.0f - (depth_ * 0.40f));
99 left = (left * dryGain) + (wetL * wetGain);
100 right = (right * dryGain) + (wetR * wetGain);
101 }

104 void updateLFOIncrement() noexcept
105 {
106 // Rango de velocidad MS2000: 0.02 Hz a 15.0 Hz en escala logarítmica
107 const float rateHz = 0.02f * abd::dsp::pow(DSPUtils::kSpeedLogScaleMax, speed_);
108 lfoInc_ = static_cast<float>(6.2831853f * rateHz / sampleRate_);
109 }

111 static inline float hermite(float frac, float y0, float y1, float y2, float y3) noexcept
112 {
113 const float c0 = y1;
114 const float c1 = 0.5f * (y2 - y0);
115 const float c2 = y0 - 2.5f * y1 + 2.0f * y2 - 0.5f * y3;
116 const float c3 = 0.5f * (y3 - y0) + 1.5f * (y1 - y2);
117 return ((c3 * frac + c2) * frac + c1) * frac + c0;
118 }
119
120 inline float readHermite(const std::array<float, kBufferSize> &buf, float delaySamples) const noexcept
121 {
122 float readPos = static_cast<float>(writeIndex_) - delaySamples;
123 if (readPos < 0.0f)
124 readPos += static_cast<float>(kBufferSize);
125
126 const int i0 = static_cast<int>(readPos) & kBufferMask;
127 const float frac = readPos - static_cast<float>(static_cast<int>(readPos));
128 const int i1 = (i0 + 1) & kBufferMask;
129 const int i2 = (i0 + 2) & kBufferMask;
130 const int i3 = (i0 + 3) & kBufferMask;
131
132 return hermite(frac, buf[i0], buf[i1], buf[i2], buf[i3]);
133 }
