108 /**
109 * Musical Bounded Randomizer Algorithm:
110 * Generates cohesive, instantly playable patches (Basses, Leads, Pads, Arps)
111 * avoiding digital silence, clicks or screeching feedback.
112 */
113 static void buildMusicalRandomPatch(juce::AudioProcessorValueTreeState& apvts)
114 {
115 juce::Random r;
116
117 // 1. Oscillators: ensure audible tone
118 setParam(apvts, ParamIDs::osc1Wave, static_cast<float>(r.nextInt(6))); // Saw, Pulse, Tri, Sin, Vox, DWGS
119 setParam(apvts, ParamIDs::osc1Ctrl1, r.nextFloat() * 127.0f);
120 setParam(apvts, ParamIDs::osc1DwgsWave, static_cast<float>(r.nextInt(64)));
121
122 setParam(apvts, ParamIDs::osc2Wave, static_cast<float>(r.nextInt(3)));
123 setParam(apvts, ParamIDs::osc2ModType, static_cast<float>(r.nextInt(4)));
124 setParam(apvts, ParamIDs::osc2Semitone, static_cast<float>(r.nextInt(25) - 12)); // +/- 12 semitones
125 setParam(apvts, ParamIDs::osc2Tune, static_cast<float>(r.nextInt(41) - 20));
126
127 // Mixer: HOT range for OSC1 (80..127), optional OSC2 and subtle noise
128 setParam(apvts, ParamIDs::mixOsc1Level, r.nextFloat() * 47.0f + 80.0f);
129 setParam(apvts, ParamIDs::mixOsc2Level, r.nextFloat() * 127.0f);
130 setParam(apvts, ParamIDs::mixNoiseLevel, r.nextFloat() * 30.0f); // Max 30 noise
131
132 // 2. Filter: Bounded sweet spot (Cutoff 40..127, Resonance <= 85)
133 setParam(apvts, ParamIDs::filterType, static_cast<float>(r.nextInt(4)));
134 setParam(apvts, ParamIDs::filterCutoff, r.nextFloat() * 87.0f + 40.0f);
135 setParam(apvts, ParamIDs::filterResonance, r.nextFloat() * 85.0f);
136 setParam(apvts, ParamIDs::filterEg1Int, static_cast<float>(r.nextInt(81) - 40));
137 setParam(apvts, ParamIDs::filterKeyTrack, static_cast<float>(r.nextInt(61) - 30));
138
139 // 3. Envelopes: Coupled ADSR to prevent zero-gating
140 setParam(apvts, ParamIDs::eg1Attack, r.nextFloat() * 70.0f);
141 setParam(apvts, ParamIDs::eg1Decay, r.nextFloat() * 90.0f + 20.0f);
142 setParam(apvts, ParamIDs::eg1Sustain, r.nextFloat() * 127.0f);
143 setParam(apvts, ParamIDs::eg1Release, r.nextFloat() * 80.0f + 10.0f);
144
145 float eg2Sustain = r.nextFloat() * 127.0f;
146 setParam(apvts, ParamIDs::eg2Sustain, eg2Sustain);
147 if (eg2Sustain < 30.0f) {
148 setParam(apvts, ParamIDs::eg2Decay, r.nextFloat() * 60.0f + 50.0f); // Extended decay
149 } else {
150 setParam(apvts, ParamIDs::eg2Decay, r.nextFloat() * 100.0f);
151 }
152 setParam(apvts, ParamIDs::eg2Attack, r.nextFloat() * 40.0f); // Snappy attack
153 setParam(apvts, ParamIDs::eg2Release, r.nextFloat() * 70.0f + 10.0f);
154
155 // 4. LFOs
156 setParam(apvts, ParamIDs::lfo1Wave, static_cast<float>(r.nextInt(4)));
157 setParam(apvts, ParamIDs::lfo1Freq, r.nextFloat() * 90.0f + 10.0f);
158 setParam(apvts, ParamIDs::lfo2Wave, static_cast<float>(r.nextInt(4)));
159 setParam(apvts, ParamIDs::lfo2Freq, r.nextFloat() * 90.0f + 10.0f);
160
161 // 5. Virtual Patch Matrix (50% probability per slot, moderate intensity -40..+40)
162 auto randomPatchSlot = [&](const char* srcId, const char* destId, const char* intId) {
163 if (r.nextBool()) {
164 setParam(apvts, srcId, static_cast<float>(r.nextInt(8)));
165 setParam(apvts, destId, static_cast<float>(r.nextInt(8)));
166 setParam(apvts, intId, static_cast<float>(r.nextInt(81) - 40));
167 } else {
168 setParam(apvts, intId, 0.0f);
169 }
170 };
171
172 randomPatchSlot(ParamIDs::patch1Source, ParamIDs::patch1Destination, ParamIDs::patch1Intensity);
173 randomPatchSlot(ParamIDs::patch2Source, ParamIDs::patch2Destination, ParamIDs::patch2Intensity);
174 randomPatchSlot(ParamIDs::patch3Source, ParamIDs::patch3Destination, ParamIDs::patch3Intensity);
175 randomPatchSlot(ParamIDs::patch4Source, ParamIDs::patch4Destination, ParamIDs::patch4Intensity);
176
177 // Subtle 15% overdrive chance
178 setParam(apvts, ParamIDs::ampDistortion, (r.nextFloat() > 0.85f) ? 1.0f : 0.0f);
179 }
