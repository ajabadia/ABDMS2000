115 /**
116 * Musical Bounded Randomizer Algorithm (adapted from ReMS2000 + musical sweet spots):
117 * Generates cohesive, instantly playable patches (Basses, Leads, Pads, Arps)
118 * avoiding digital silence (e.g. HPF cutoff clamping, AudioIn skip), clicks, or screeching feedback.
119 */
120 static void buildMusicalRandomPatch(juce::AudioProcessorValueTreeState &apvts)
121 {
122 // Use deterministic LCG for WASM compatibility (juce::Random default constructor
123 // seeks system entropy which is unavailable in AudioWorklet/WASM).
124 uint32_t seed = static_cast<uint32_t>(juce::Time::getMillisecondCounterHiRes() * 1000);
125 auto nextFloat = [&seed]() -> float
126 {
127 seed = seed * 1664525u + 1013904223u;
128 return static_cast<float>(seed & 0xFFFFFF) / 16777216.0f;
129 };
130 auto nextInt = [&seed](int maxExclusive) -> int
131 {
132 seed = seed * 1664525u + 1013904223u;
133 return static_cast<int>(seed % static_cast<uint32_t>(maxExclusive));
134 };
135 auto nextRangeInt = [&nextInt](int minInclusive, int maxInclusive) -> int
136 { return minInclusive + nextInt(maxInclusive - minInclusive + 1); };
137 auto nextBool = [&seed]() -> bool
138 {
139 seed = seed * 1664525u + 1013904223u;
140 return (seed & 0x80000000) != 0;
141 };

143 auto randomizeTimbre = [&](const juce::String &prefix)
144 {
145 auto pid = [&](const char *base) -> juce::String
146 {
147 if (prefix.isEmpty())
148 return juce::String(base);
149 juce::String s(base);
150 return prefix + s.substring(0, 1).toUpperCase() + s.substring(1);
151 };
152 auto setTParam = [&](const char *base, float val)
153 {
154 const auto id = pid(base);
155 setParam(apvts, id.toRawUTF8(), val);
156 };

158 // 1. Voice Assign, Detune & Portamento (ReMS2000 bounds)
159 setTParam("voiceMode", static_cast<float>(nextRangeInt(0, 2))); // Mono, Poly, Unison
160 setTParam("unisonDetune", static_cast<float>(nextRangeInt(5, 45))); // Musical unison detune (5..45 cents)
161 setTParam("portamentoTime", nextBool() ? static_cast<float>(nextRangeInt(0, 45)) : 0.0f);
162
163 // 2. Oscillators: skip AudioIn (7) per ReMS2000 rndValueMan(0, 6); musical semitone intervals
164 setTParam("osc1Wave", static_cast<float>(nextRangeInt(0, 6))); // Saw..Noise (skip AudioIn)
165 setTParam("osc1Ctrl1", static_cast<float>(nextRangeInt(0, 127)));
166 setTParam("osc1Ctrl2", static_cast<float>(nextRangeInt(0, 127)));
167 setTParam("osc1DwgsWave", static_cast<float>(nextRangeInt(0, 63)));
168
169 setTParam("osc2Wave", static_cast<float>(nextRangeInt(0, 2)));
170 const int osc2Mod = (nextFloat() < 0.65f) ? 0 : nextRangeInt(1, 3);
171 setTParam("osc2ModType", static_cast<float>(osc2Mod));
172
173 static constexpr int kMusicalIntervals[] = {-12, -7, -5, 0, 0, 7, 12};
174 const int semi = (osc2Mod == 0) ? kMusicalIntervals[nextInt(7)] : nextRangeInt(-12, 12);
175 setTParam("osc2Semitone", static_cast<float>(semi));
176 setTParam("osc2Tune", static_cast<float>(nextRangeInt(-15, 15)));
177
178 // 3. Mixer: ReMS2000 guarantees audible OSC1/OSC2 (45..127) and bounded Noise (* 0.5)
179 setTParam("mixOsc1Level", static_cast<float>(nextRangeInt(65, 127)));
180 setTParam("mixOsc2Level", static_cast<float>(nextRangeInt(45, 127)));
181 setTParam("mixNoiseLevel", static_cast<float>(nextRangeInt(0, 45)));
182
183 // 4. Filter: ReMS2000 HPF rule (if filterType == 3 HPF12, clamp cutoff to 0..40 to avoid thin silence)
184 const int fType = nextRangeInt(0, 3);
185 setTParam("filterType", static_cast<float>(fType));
186 if (fType == 3)
187 setTParam("filterCutoff", static_cast<float>(nextRangeInt(0, 40)));
188 else
189 setTParam("filterCutoff", static_cast<float>(nextRangeInt(35, 127)));
190
191 setTParam("filterResonance", static_cast<float>(nextRangeInt(0, 38))); // ReMS2000: * 0.3
192 setTParam("filterEg1Int", static_cast<float>(nextRangeInt(-25, 25))); // ReMS2000: * 0.4
193 setTParam("filterKeyTrack", static_cast<float>(nextRangeInt(-30, 30)));
194 setTParam("filterVelo", static_cast<float>(nextRangeInt(-30, 30)));

242 setTParam("modSeqOn", (nextFloat() < 0.35f) ? 1.0f : 0.0f);
243 setTParam("modSeqType", static_cast<float>(nextRangeInt(0, 3)));
244 setTParam("modSeqResolution", static_cast<float>(nextRangeInt(2, 8)));

246 setTParam("seqLastStep", static_cast<float>(kStepLengths[nextInt(4)]));
247 setTParam("seqRunMode", static_cast<float>(nextRangeInt(0, 1)));
248 setTParam("seqKeySync", static_cast<float>(nextRangeInt(0, 2)));

250 for (int lane = 1; lane <= 3; ++lane)

323 /**
324 * Swaps all 115 parameters between Timbre 1 and Timbre 2 (`<param>` <-> `t2<Param>`).
325 */
326 static void swapTimbres(juce::AudioProcessorValueTreeState &apvts)
327 {
328 forEachTimbrePair(
329 [&](const juce::String &t1Id, const juce::String &t2Id)
330 {
331 auto *p1 = apvts.getRawParameterValue(t1Id);
332 auto *p2 = apvts.getRawParameterValue(t2Id);
333 if (p1 != nullptr && p2 != nullptr)
334 {
335 const float v1 = p1->load();
336 const float v2 = p2->load();
337 setParam(apvts, t1Id.toRawUTF8(), v2);
338 setParam(apvts, t2Id.toRawUTF8(), v1);
339 }
340 });
341 }

343 private:
344 template <typename PairFn> static void forEachTimbrePair(PairFn &&fn)
345 {
346 for (const auto &meta : ParameterRegistry::getAllParameters())
347 {
348 const juce::String id(meta.id);
349 if (id.startsWith("t2") && id.length() > 2)
350 {
351 const juce::String t1Id = id.substring(2, 3).toLowerCase() + id.substring(3);
352 if (ParameterRegistry::getParameter(t1Id.toRawUTF8()) != nullptr)
353 fn(t1Id, id);
354 }
355 }
