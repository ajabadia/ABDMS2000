172/**
173 * @brief Factor de aceleración de LFO2 respecto a LFO1 extraído de las tablas ROM
174 * 0x295E8 (LFO1) y 0x29BE8 (LFO2).
175 *
176 * Para valores de knob 0..114 ambos LFOs tienen exactamente el mismo incremento
177 * (ratio = 1.0f; última palabra común 0x00207A en 114). La primera diferencia es
178 * 115 (LFO1=0x0021A0, LFO2=0x0020D4); LFO2 queda por debajo en 115..120,
179 * vuelve a superarlo en 121 y alcanza una razón de 5.3239x en 127.
180 */
181inline float getHardwareLfo2SpeedRatio(float midiKnob0to127) noexcept
182{
183 const float clamped = std::clamp(midiKnob0to127, 0.0f, 127.0f);
184 const int i0 = static_cast<int>(clamped);
185 const int i1 = std::min(127, i0 + 1);
186 const float frac = clamped - static_cast<float>(i0);
187
188 const float r0 = static_cast<float>(kRomLfo2IncrementQ23[i0]) / static_cast<float>(kRomLfo1IncrementQ23[i0]);
189 const float r1 = static_cast<float>(kRomLfo2IncrementQ23[i1]) / static_cast<float>(kRomLfo1IncrementQ23[i1]);
190 return r0 + frac * (r1 - r0);
191}

235 * el límite que el microcódigo aplica en P:$0760..$0766 (`#>$77` en Q7).

239/**
240 * @brief Frecuencia de corte exacta en Hz del VCF tal como la implementa el hardware.
241 *
242 * El H8S **no** envía el cutoff desde `kRomFilterCutoffQ23` (0x26154): el valor que escribe
243 * al DSP es la tabla maestra de pitch, `kRomMasterPitchQ23[128 + knob]` (0x240F0).
244 * Verificado por volcado de X: del DSP56362 en ms2000r-emu (7 knobs, 0x4B0 = word exacto,
245 * idéntico en las 4 voces e independiente de la resonancia):
246 *
247 * X:$04B0(voice i) = kRomMasterPitchQ23[128 + knob]
248 *
249 * El microcódigo (P:$0740..$078F) convierte ese word en el coeficiente de la cámara resonante
250 * y lo recorta en `$770000` (`kHardwareFilterMaxCoefficient`, 0.9297). La relación exacta,
251 * medida por autoscilación del filtro en el emulador (reso 127, cola tras note-off) en los
252 * knobs 20/60/90/100/127, es un coeficiente tipo Chamberlin:
253 *
254 * c = min(2 * w, 0.9297) con w = word / 2^23
255 * c = sin(pi * fc / fs) => fc = (fs / pi) * asin(c)
256 *
257 * Error de la ley frente a la autoscilación medida: 0.14 / 0.13 / 0.15 / 0.18 / 0.28 %
258 * (207.5 / 2090 / 9348 / 13696 / 18186 Hz). Las leyes de tipo `tan` (pre-warping bilineal)
259 * fallan en el extremo alto: `atan(2w)` queda 26 % bajo en el knob 127 porque el hardware
260 * NO pre-warpéa, satura en 0.9297.
261 *
262 * Consecuencias de la ley:
263 * - knob 0 = 65.4 Hz, 1 semitono por paso hasta ~knob 72 (ahí empieza la compresión de la sin).
264 * - El techo de 0.9297 se alcanza en el knob ~109: de ahí en adelante fc = 18,193 Hz.
265 * - Por debajo del knob 0 la misma tabla (índices 0..127) sigue bajando hasta ~0.2 Hz para
266 * modulación negativa.
267 *
268 * `modOctaves` replica el bloque `P:$0749..$0768`:
269 * - `X:$04B0` (`wStatic`) se obtiene de `kRomMasterPitchQ23[128 + knob + kbdSemis]` (`H8S 0x13390`).
270 * - La modulación dinámica (`EG1`, `VirtualPatch`, `FilterVelo`) se recorta a `[-8.0, +8.0]` octavas
271 * (`P:$074D..$074F`: `add #$40,a` + clamp `[0, $7FFFFF]`) y se convierte en el multiplicador lineal
272 * `2^modOctaves` (`P:$0751..$075B`) que multiplica a `wStatic` (`P:$075E..$075F`) antes de saturar en `$770000`.
273 */
274inline float getHardwareFilterCutoffHz(float cutoffNorm0to1,
275 float modOctaves = 0.0f,
276 float w3Scale = 1.0f,
277 float kbdOctaves = 0.0f) noexcept
278{
279 const float knob0to127 = std::clamp(cutoffNorm0to1, 0.0f, 1.0f) * 127.0f;
280 const float staticRomIndex = std::clamp(128.0f + knob0to127 + (kbdOctaves * 12.0f), 0.0f, 255.0f);
281 const int i0 = static_cast<int>(staticRomIndex);
282 const int i1 = std::min(255, i0 + 1);
283 const float frac = staticRomIndex - static_cast<float>(i0);
284
285 const double w0 = static_cast<double>(kRomMasterPitchQ23[i0]);
286 const double w1 = static_cast<double>(kRomMasterPitchQ23[i1]);
287 const double wStatic = (w0 + static_cast<double>(frac) * (w1 - w0)) / kQ23FractionOne;
288
289 const double clampedModOctaves = std::clamp(static_cast<double>(modOctaves), -8.0, 8.0);
290 const double w = wStatic * std::exp2(clampedModOctaves);
291
292 const double c = std::min(2.0 * w, kHardwareFilterMaxCoefficient) * static_cast<double>(w3Scale);
293 const double hz = (static_cast<double>(kHardwareSampleRate) / 3.14159265358979323846) * std::asin(std::clamp(c, 0.0, 0.999));
294 return static_cast<float>(std::clamp(hz, 5.0, 20000.0));
295}

776/**
777 * @brief Devuelve el factor de pendiente anti-aliasing S (`X:$00DB` para OSC1, `X:$00DD` para OSC2)
778 * interpolando linealmente `Y:$0800..$087F` (`isOsc2 = false`) o `Y:$0881..$08FF` (`isOsc2 = true`)
779 * según `P:$0646..$0659` para la frecuencia `freqHz` a la frecuencia de muestreo `sampleRate`.
780 */
781inline float getHardwareOscSlopeFactor(float freqHz, bool isOsc2 = false,
782 double sampleRate = kHardwareSampleRate) noexcept
783{
784 const double sr = std::max(1000.0, sampleRate);
785 // Normalizamos a la frecuencia equivalente a 48 kHz (la tabla depende de f0 / fs)
786 const double equivFreq48k = std::max(1.0, static_cast<double>(freqHz) * (static_cast<double>(kHardwareSampleRate) / sr));
787 const double midiPitch = 69.0 + 12.0 * std::log2(equivFreq48k / 440.0);
788 // En P:$062A, OSC1 carga n5 = $0800 (índice = midiPitch) y OSC2 carga n5 = $0881 (índice = midiPitch + 1 dentro de Y:$0880)
789 const double tablePos = isOsc2 ? std::clamp(midiPitch + 1.0, 1.0, 127.0)
790 : std::clamp(midiPitch, 0.0, 127.0);
791 const int i0 = static_cast<int>(tablePos);
792 const int i1 = std::min(127, i0 + 1);
793 const double frac = tablePos - static_cast<double>(i0);
794 const int32_t *table = isOsc2 ? kRomOsc2SlopeQ23 : kRomOsc1SlopeQ23;
795 const double s0 = static_cast<double>(table[i0]) / kQ23FractionOne;
796 const double s1 = static_cast<double>(table[i1]) / kQ23FractionOne;
797 return static_cast<float>(std::clamp(s0 + frac * (s1 - s0), 0.0, 1.0));
798}
