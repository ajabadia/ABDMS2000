22/**
23 * @brief Formato SysEx nativo y binario del Korg microKORG físico (254 bytes decodificados).
24 *
25 * Bifurcación especializada del programa de hardware que modela las características
26 * exclusivas del microKORG frente al MS2000:
27 * - Byte 0x25: Arpeggiator Last Step (longitud rítmica de 1 a 8 pasos).
28 * - Byte 0x0C: Arpeggiator Step Mask (bitmask de 8 bits donde bit 0..7 representan
29 * el estado Play/Rest de los pasos 1 a 8).
30 * - Neutralización del Modulation Sequencer (el microKORG no dispone de secuenciador físico).
31 * - Identificador de modelo Korg: soporta 0x58 (compatibilidad estándar) y 0x71 (microKORG nativo).
32 */
33struct MicroKorgHardwareProgram
34{
35 static constexpr size_t PROGRAM_SIZE = 254;
36 static constexpr size_t NAME_OFFSET = 0x00;
37 static constexpr size_t NAME_LENGTH = 12;
38
39 // Bytes exclusivos de arpegiador microKORG
40 static constexpr size_t ARP_STEP_MASK = 0x0C; // Bitmask 8 bits (Step 1..8 Play/Rest)
41 static constexpr size_t ARP_LAST_STEP = 0x25; // Longitud de 1 a 8 pasos
42
43 // Sub-status de comandos SysEx Korg
44 static constexpr uint8_t CMD_PROGRAM_DUMP = 0x40;
45 static constexpr uint8_t CMD_ALL_DATA_DUMP = 0x4C;
46 static constexpr uint8_t CMD_WRITE_COMPLETED = 0x23;
47 static constexpr uint8_t CMD_WRITE_ERROR = 0x24;

49 // Model IDs
50 static constexpr uint8_t MODEL_ID_MICROKORG = 0x71;

72 std::string getName() const

106 bool getStepMask(int stepIndex1to8) const noexcept
107 {
108 if (stepIndex1to8 < 1 || stepIndex1to8 > 8)
109 return true;
110 const uint8_t mask = static_cast<uint8_t>(1u << (stepIndex1to8 - 1));

112 }

114 void setStepMask(int stepIndex1to8, bool active) noexcept
115 {
116 if (stepIndex1to8 < 1 || stepIndex1to8 > 8)
117 return;
118 const uint8_t mask = static_cast<uint8_t>(1u << (stepIndex1to8 - 1));
119 if (active)

135 /** Desempaqueta 254 B de un payload 7→8 de microKORG. */
136 bool unpackFromSysexPayload(const uint8_t *payload, size_t payloadLen)
137 {
138 std::vector<uint8_t> unpacked;
139 if (!SysExCodec::unpack7to8(payload, payloadLen, unpacked) || unpacked.empty())
140 return false;
141 if (unpacked.size() < PROGRAM_SIZE)
142 return false;
143 std::copy_n(unpacked.begin(), PROGRAM_SIZE, raw.begin());
144 return true;
145 }
146
147 /** Empaqueta los 254 B al payload 7→8 de Korg. */
148 bool packToSysexPayload(std::vector<uint8_t> &outPayload) const
149 {
150 return SysExCodec::pack8to7(raw.data(), PROGRAM_SIZE, outPayload);
151 }

153 /** Construye la trama SysEx completa para microKORG (F0 42 3n modelID 40 [payload] F7). */
154 std::vector<uint8_t>
155 buildProgramDump(int channel = 1, uint8_t modelId = MODEL_ID_MICROKORG, uint8_t function = CMD_PROGRAM_DUMP) const
156 {
157 std::vector<uint8_t> packed;
158 if (!packToSysexPayload(packed))
159 return {};
160
161 std::vector<uint8_t> sysex;
162 sysex.reserve(6 + packed.size());
163 sysex.push_back(0xF0);
164 sysex.push_back(0x42); // Korg ID
165 sysex.push_back(MS2000HardwareProgram::korgChannelByte(channel));
166 sysex.push_back(modelId); // 0x71 o 0x58
167 sysex.push_back(function); // 0x40
168 sysex.insert(sysex.end(), packed.begin(), packed.end());
169 sysex.push_back(0xF7);
170 return sysex;
171 }

174 /**
175 * Vuelca los valores del volcado microKORG hacia el árbol de parámetros del motor (APVTS).
176 */
177 void applyToAPVTS(juce::AudioProcessorValueTreeState &apvts) const
178 {
179 // 1. Aplicar la parte compartida del motor MS2000/microKORG
180 MS2000HardwareProgram base;
181 base.raw = raw;
182 base.applyToAPVTS(apvts);
183
184 // 2. Aplicar los parámetros exclusivos de máscara y longitud rítmica del arpegiador
185 auto setParam = [&apvts](const char *paramId, float val)
186 {
187 if (auto *p = apvts.getParameter(paramId))
188 p->setValueNotifyingHost(p->convertTo0to1(val));
189 };
190
191 const int lastStep = getLastStep();
192 setParam("arpLastStep", static_cast<float>(lastStep));
193
194 for (int s = 1; s <= 8; ++s)
195 {
196 const std::string id = "arpStep" + std::to_string(s);
197 setParam(id.c_str(), getStepMask(s) ? 1.0f : 0.0f);
198 }
199
200 // 3. Inhibir el mod sequencer si estuviera activo (paridad microKORG)
201 setParam("modSeqOn", 0.0f);
202 }

204 /**
205 * Extrae el estado actual del motor (APVTS) hacia la estructura microKORG.

207 void captureFromAPVTS(const juce::AudioProcessorValueTreeState &apvts, const std::string &name = {})
208 {
209 // 1. Extraer los parámetros comunes de síntesis
210 MS2000HardwareProgram base;
211 base.raw = raw;
212 base.captureFromAPVTS(apvts, name);
213 raw = base.raw;
214
215 // 2. Extraer los parámetros exclusivos del arpegiador microKORG
216 auto getParam = [&apvts](const char *paramId, float defaultVal) -> float
217 {
218 if (auto *p = apvts.getParameter(paramId))
219 return p->convertFrom0to1(p->getValue());
220 return defaultVal;
221 };
222
223 const int lastStep = static_cast<int>(std::round(getParam("arpLastStep", 8.0f)));
224 setLastStep(lastStep);
225
226 uint8_t maskByte = 0;
227 for (int s = 1; s <= 8; ++s)
228 {
229 const std::string id = "arpStep" + std::to_string(s);
230 const float val = getParam(id.c_str(), 1.0f);
231 if (val > 0.5f)
232 maskByte |= static_cast<uint8_t>(1u << (s - 1));
233 }
234 setStepMaskByte(maskByte);
235 }
